# The Waffen SS Infanterie Division (common/scripted_effects/wif_ger_himmler.txt)
# is built from militia battalions. Vanilla's GER_SS_01 only accepts infantry,
# motorized, medium_armor and mechanized, so it rejects that template and the
# divisions fall back to generic names. This is the same list opened to militia,
# numbered together with GER_SS_01 so the two never hand out the same number.
WIF_GER_SS_MIL = {
	name = "Waffen-SS Divisions"

	# OSB: the Ordensstaat Burgund raises the SS again when Himmler takes it out
	# of the Reich (common/scripted_effects/wif_ger_himmler.txt).
	for_countries = { GER OSB }

	can_use = {
		always = yes
	}

	division_types = { "militia" "infantry" }

	link_numbering_with = { GER_SS_01 }

	ordered = {

# The Waffen SS Panzer Division (common/scripted_effects/wif_ger_himmler.txt).
# Numbered together with the infantry and with vanilla's GER_SS_01, so no two
# SS divisions ever share a number.
WIF_GER_SS_ARM = {
	name = "Waffen-SS Panzer Divisions"

	for_countries = { GER OSB }

	can_use = {
		always = yes
	}

	division_types = { "medium_armor" "heavy_armor" }

	link_numbering_with = { GER_SS_01 WIF_GER_SS_MIL }

	fallback_name = "%d. SS-Panzer-Division"
}
