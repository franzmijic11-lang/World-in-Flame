# Allied AI air forces yield crowded airfields to players.
#
# The problem: an allied AI can fill every airfield in a region with its own
# wings, leaving a player nowhere to base theirs. Script cannot move or
# disband an air wing, so the AI is steered instead, the way Frontline AI
# evacuates regions: a strategy with strategic_air_importance -500000 for the
# region, which makes the AI rebase its planes elsewhere.
#
# Once a week, for each player (on_weekly, common/on_actions/wif_on_actions.txt):
#   - for every strategic region, add up the air base capacity of the states
#     there held by the player or an ally (wif_air_friend_of_root), at 200
#     planes a level, and the planes those countries have based in it;
#   - at over 90% of capacity the region is crowded (player flag
#     wif_air_crowded_<region>). It stays crowded until it falls under 70%,
#     and for at least 30 days (wif_air_hold_<region>), so the AI is not
#     shuffled back and forth every week;
#   - while it is crowded, every allied AI gets a 10-day flag
#     wif_air_yield_<region>, refreshed weekly, which switches on its strategy
#     in common/ai_strategy/wif_air_yield_auto.txt.
#
# Regions where there is room are left alone - allied AIs base wherever they
# like until airfields fill up. Works per region, not per airfield: one full
# airfield next to an empty one does not count as crowded.
#
# The per-region code is generated by tools/gen_air_yield.py.


# Player scope. Every country that can share the player's airfields,
# including the player.
wif_air_build_friends = {
	clear_array = wif_air_friends
	every_country = {
		limit = { wif_air_friend_of_root = yes }
		add_to_array = { ROOT.wif_air_friends = THIS }
	}
}

# Player scope, with wif_air_region set: wif_air_used = the planes every
# friend has based in that region.
wif_air_count_region = {
	set_variable = { wif_air_used = 0 }
	for_each_scope_loop = {
		array = wif_air_friends
		wif_air_count_planes = yes
	}
}

# Country scope, ROOT = the player. Adds the planes this country has based in
# region ROOT.wif_air_region to ROOT.wif_air_used.
#
# num_planes_stationed_in_regions only compares against a number written in
# the script, so the count is found by binary search, building each comparison
# with meta_trigger. 14 steps count exactly up to 16383 planes.
wif_air_count_planes = {
	if = {
		limit = {
			meta_trigger = {
				text = { num_planes_stationed_in_regions = { value > 0 regions = { [R] } } }
				R = "[?ROOT.wif_air_region|0]"
			}
		}
		set_variable = { ROOT.wif_air_count = 0 }
		set_variable = { ROOT.wif_air_step = 8192 }
		while_loop_effect = {
			limit = { ROOT = { check_variable = { wif_air_step > 0.5 } } }

			# Is the count at least count + step?
			set_variable = { ROOT.wif_air_probe = ROOT.wif_air_count }
			add_to_variable = { ROOT.wif_air_probe = ROOT.wif_air_step }
			subtract_from_variable = { ROOT.wif_air_probe = 1 }
			if = {
				limit = {
					meta_trigger = {
						text = { num_planes_stationed_in_regions = { value > [V] regions = { [R] } } }
						V = "[?ROOT.wif_air_probe|0]"
						R = "[?ROOT.wif_air_region|0]"
					}
				}
				add_to_variable = { ROOT.wif_air_count = ROOT.wif_air_step }
			}
			divide_variable = { ROOT.wif_air_step = 2 }
		}
		add_to_variable = { ROOT.wif_air_used = ROOT.wif_air_count }
	}
}
