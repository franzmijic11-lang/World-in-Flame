# The Scottish campaign against Fascist Britain. Fired by the decision
# wif_sco_terror_strike (common/decisions/wif_peace_deal.txt).


# State scope: the target state. ROOT is Scotland.
#
# Base chance 90%, minus 5% for every British division with military police
# in the state. A success wrecks 1 to 3 buildings, picked at random from what
# the state actually has.
wif_sco_terror_strike = {
	save_global_event_target_as = wif_sco_strike_target

	set_temp_variable = { wif_sco_mp = 0 }
	every_state_division = {
		limit = {
			OWNER = { original_tag = ENG }
			division_has_battalion_in_template = military_police
		}
		add_to_temp_variable = { wif_sco_mp = 1 }
	}
	set_temp_variable = { wif_sco_cut = wif_sco_mp }
	multiply_temp_variable = { wif_sco_cut = 5 }
	set_temp_variable = { wif_sco_chance = 90 }
	subtract_from_temp_variable = { wif_sco_chance = wif_sco_cut }
	clamp_temp_variable = {
		var = wif_sco_chance
		min = 0
		max = 90
	}
	ROOT = {
		set_variable = { wif_sco_last_chance = wif_sco_chance }
		set_variable = { wif_sco_last_mp = wif_sco_mp }
	}

	set_temp_variable_to_random = {
		var = wif_sco_roll
		min = 0
		max = 100
	}

	if = {
		limit = {
			check_variable = { wif_sco_roll < wif_sco_chance }
		}
		set_temp_variable_to_random = {
			var = wif_sco_n
			min = 1
			max = 4
			integer = yes
		}
		set_temp_variable = { wif_sco_done = 0 }
		while_loop_effect = {
			limit = {
				check_variable = { wif_sco_n > 0 }
				wif_sco_state_has_a_target = yes
			}
			random_list = {
				20 = {
					modifier = {
						factor = 0
						NOT = { check_variable = { building_level@infrastructure > 0 } }
					}
					remove_building = {
						type = infrastructure
						level = 1
					}
				}
				20 = {
					modifier = {
						factor = 0
						NOT = { check_variable = { building_level@industrial_complex > 0 } }
					}
					remove_building = {
						type = industrial_complex
						level = 1
					}
				}
				20 = {
					modifier = {
						factor = 0
						NOT = { check_variable = { building_level@arms_factory > 0 } }
					}
					remove_building = {
						type = arms_factory
						level = 1
					}
				}
				15 = {
					modifier = {
						factor = 0
						NOT = { check_variable = { building_level@synthetic_refinery > 0 } }
					}
					remove_building = {
						type = synthetic_refinery
						level = 1
					}
				}
				10 = {
					modifier = {
						factor = 0
						NOT = { check_variable = { building_level@fuel_silo > 0 } }
					}
					remove_building = {
						type = fuel_silo
						level = 1
					}
				}
				10 = {
					modifier = {
						factor = 0
						NOT = { check_variable = { building_level@dockyard > 0 } }
					}
					remove_building = {
						type = dockyard
						level = 1
					}
				}
				10 = {
					modifier = {
						factor = 0
						NOT = { check_variable = { building_level@air_base > 0 } }
					}
					remove_building = {
						type = air_base
						level = 1
					}
				}
			}
			subtract_from_temp_variable = { wif_sco_n = 1 }
			add_to_temp_variable = { wif_sco_done = 1 }
		}
		ROOT = {
			set_variable = { wif_sco_last_destroyed = wif_sco_done }
			country_event = { id = wif_peace_deal.2 }
		}
		ENG = {
			set_variable = { wif_sco_last_destroyed = wif_sco_done }
			country_event = { id = wif_peace_deal.4 }
		}
	}
	else = {
		ROOT = { country_event = { id = wif_peace_deal.3 } }
		ENG = { country_event = { id = wif_peace_deal.5 } }
	}
}
