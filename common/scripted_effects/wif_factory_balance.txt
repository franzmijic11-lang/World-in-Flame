# Military-to-civilian factory ratio cap, scaled by ideology and world tension.
#
# The permitted number of military factories is expressed as a ratio against the
# country's civilian factories, written numerator/denominator so it maps onto
# the design table directly (2-1 means two military per one civilian):
#
#   World tension   Fascism   Communism   Non-aligned   Democratic
#   ------------------------------------------------------------------
#   0%              2-1       1-1         1-2           1-3
#   25%             2-1       2-1         1-1           1-2
#   50%             3-1       3-1         2-1           2-1
#   75%             4-1       4-1         3-1           3-1
#   100%            5-1       5-1         5-1           5-1
#
# Civilian factories are never capped. Only military factories are.

wif_update_factory_cap = {
	# Default covers non-aligned at the lowest tension tier.
	set_variable = { wif_ratio_mil = 1 }
	set_variable = { wif_ratio_civ = 2 }

	if = {
		limit = { threat > 0.999 }			# 100% world tension
		set_variable = { wif_ratio_mil = 5 }
		set_variable = { wif_ratio_civ = 1 }
	}
	else_if = {
		limit = { threat > 0.7499 }			# 75%+
		if = {
			limit = { OR = { has_government = fascism has_government = communism } }
			set_variable = { wif_ratio_mil = 4 }
			set_variable = { wif_ratio_civ = 1 }
		}
		else = {
			set_variable = { wif_ratio_mil = 3 }
			set_variable = { wif_ratio_civ = 1 }
		}
	}
	else_if = {
		limit = { threat > 0.4999 }			# 50%+
		if = {
			limit = { OR = { has_government = fascism has_government = communism } }
			set_variable = { wif_ratio_mil = 3 }
			set_variable = { wif_ratio_civ = 1 }
		}
		else = {
			set_variable = { wif_ratio_mil = 2 }
			set_variable = { wif_ratio_civ = 1 }
		}
	}
	else_if = {
		limit = { threat > 0.2499 }			# 25%+
		if = {
			limit = { OR = { has_government = fascism has_government = communism } }
			set_variable = { wif_ratio_mil = 2 }
			set_variable = { wif_ratio_civ = 1 }
		}
		else_if = {
			# Democracies that have thrown off peacetime restraint fall through
			# to the non-aligned branch below instead.
			limit = {
				has_government = democratic
				NOT = { wif_uses_non_aligned_factory_cap = yes }
			}
			set_variable = { wif_ratio_mil = 1 }
			set_variable = { wif_ratio_civ = 2 }
		}
		else = {							# non-aligned
			set_variable = { wif_ratio_mil = 1 }
			set_variable = { wif_ratio_civ = 1 }
		}
	}
	else = {								# below 25%
		if = {
			limit = { has_government = fascism }
			set_variable = { wif_ratio_mil = 2 }
			set_variable = { wif_ratio_civ = 1 }
		}
		else_if = {
			limit = { has_government = communism }
			set_variable = { wif_ratio_mil = 1 }
			set_variable = { wif_ratio_civ = 1 }
		}
		else_if = {
			limit = {
				has_government = democratic
				NOT = { wif_uses_non_aligned_factory_cap = yes }
			}
			set_variable = { wif_ratio_mil = 1 }
			set_variable = { wif_ratio_civ = 3 }
		}
		else = {							# non-aligned
			set_variable = { wif_ratio_mil = 1 }
			set_variable = { wif_ratio_civ = 2 }
		}
	}

	set_variable = { wif_mil_cap = num_of_civilian_factories }
	multiply_variable = { wif_mil_cap = wif_ratio_mil }
	divide_variable = { wif_mil_cap = wif_ratio_civ }

	# Starting industry is grandfathered. A country that begins the game over
	# its ratio keeps every factory it started with - the cap only ever bites
	# on military factories built after game start.
	if = {
		limit = { check_variable = { wif_mil_cap < wif_mil_grandfather } }
		set_variable = { wif_mil_cap = wif_mil_grandfather }
	}

	# Spare capacity, read by the AI strategies in
	# common/ai_strategy/wif_factory_balance.txt. Negative or zero means the
	# country is at its ceiling and should be building civilian industry, which
	# is what raises the ceiling.
	set_variable = { wif_mil_headroom = wif_mil_cap }
	subtract_from_variable = { wif_mil_headroom = num_of_military_factories }

	# Overbuild thresholds: 25% over the ceiling is "far over", 40% is severe.
	set_variable = { wif_cap_over25 = wif_mil_cap }
	multiply_variable = { wif_cap_over25 = 1.25 }
	set_variable = { wif_cap_over40 = wif_mil_cap }
	multiply_variable = { wif_cap_over40 = 1.4 }

	# A nation at war may build without limit. The flag is refreshed weekly for
	# as long as the war lasts and lives 30 days beyond the last refresh, so the
	# ceiling returns one month after peace.
	#
	# It was 90 days. Three months proved long enough that a country could finish
	# a war, sit on an enormous overbuild for a full season and never once meet
	# the penalty it had earned, which made the ceiling feel absent rather than
	# merely deferred. One month is what the battalion cap already allows for
	# demobilisation, so the two systems now agree on what a reasonable wind-down
	# looks like.
	#
	# The clock is keyed on the LAST WEEKLY TICK SPENT AT WAR, not on the peace
	# deal, so it runs one month from the last tick of fighting rather than one
	# month from the treaty. Those are within seven days of each other.
	if = {
		limit = { has_war = yes }
		set_country_flag = { flag = wif_cap_war_grace days = 30 value = 1 }
	}

	# One-time fix-up for campaigns saved while the grace was still 90 days.
	#
	# Shortening the number above only takes effect the next time the flag is
	# SET, and it is only ever set while at war - so a country already at peace
	# would quietly run out the old ninety day clock and never see the change.
	# Re-issuing the flag overwrites that timer, giving one month from now
	# instead of up to three.
	#
	# wif_cap_grace_migrated is set unconditionally the first time a country
	# reaches this line, so the re-issue happens at most once and can never hand
	# a country still at war a second helping on top of its own weekly refresh.
	# Safe to delete once no pre-change save is in use.
	if = {
		limit = { NOT = { has_country_flag = wif_cap_grace_migrated } }
		set_country_flag = wif_cap_grace_migrated

		if = {
			limit = {
				has_war = no
				has_country_flag = wif_cap_war_grace
			}
			set_country_flag = { flag = wif_cap_war_grace days = 30 value = 1 }
		}
	}

	if = {
		limit = { has_country_flag = wif_cap_war_grace }
		wif_clear_ceiling_spirit = yes
	}
	else_if = {
		limit = {
			check_variable = { num_of_military_factories > wif_cap_over40 }
			NOT = { has_idea = wif_military_factory_ceiling_severe }
		}
		wif_clear_ceiling_spirit = yes
		add_ideas = wif_military_factory_ceiling_severe
	}
	else_if = {
		limit = {
			check_variable = { num_of_military_factories > wif_cap_over25 }
			NOT = { check_variable = { num_of_military_factories > wif_cap_over40 } }
			NOT = { has_idea = wif_military_factory_ceiling_far }
		}
		wif_clear_ceiling_spirit = yes
		add_ideas = wif_military_factory_ceiling_far
	}
	else_if = {
		limit = {
			check_variable = { wif_mil_headroom < 1 }
			NOT = { check_variable = { num_of_military_factories > wif_cap_over25 } }
			NOT = { has_idea = wif_military_factory_ceiling }
		}
		wif_clear_ceiling_spirit = yes
		add_ideas = wif_military_factory_ceiling
	}
	else_if = {
		limit = { check_variable = { wif_mil_headroom > 0 } }
		wif_clear_ceiling_spirit = yes
	}
}

# Strips whichever ceiling tier is currently applied. Guarded so a country that
# has none is left untouched rather than churning ideas every week.
wif_clear_ceiling_spirit = {
	if = {
		limit = { has_idea = wif_military_factory_ceiling }
		remove_ideas = wif_military_factory_ceiling
	}
	if = {
		limit = { has_idea = wif_military_factory_ceiling_far }
		remove_ideas = wif_military_factory_ceiling_far
	}
	if = {
		limit = { has_idea = wif_military_factory_ceiling_severe }
		remove_ideas = wif_military_factory_ceiling_severe
	}
}
