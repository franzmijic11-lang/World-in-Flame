# Settlement terms for a country taking Germany's separate peace.
# Run in the scope of the country that is settling.
#
# White peace is made with Germany AND every country in a faction with Germany,
# not just Germany itself, so a co-belligerent cannot keep the war alive on its
# own. Inside every_country the scope is the Axis member and PREV is the
# settling country, which is the form white_peace documents and vanilla uses.
wif_settle_with_axis = {
	leave_faction = yes

	every_country = {
		limit = {
			OR = {
				tag = GER
				is_in_faction_with = GER
			}
			has_war_with = PREV
		}
		white_peace = PREV
	}

	add_timed_idea = {
		idea = wif_negotiated_settlement
		days = 1825
	}
}
