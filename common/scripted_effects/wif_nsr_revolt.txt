# The Novgorod rising.
#
# Deliberately NOT start_civil_war: that would split Moskowien's army between
# the two sides. Moskowien keeps every division it has, and the rebels get
# purpose-built formations of their own from NSR_revolt.txt instead.
#
# Run in RKM's scope.
wif_nsr_revolt = {
	# Marks the civil war as running. Cleared when Moskowien has put it down,
	# which is what triggers Saint Petersburg folding back in.
	set_country_flag = wif_moskowien_civil_war

	# Everything below strips ~132 victory points of territory from Moskowien in
	# one go, 35 of it to the loyalist emergency governments. Capitulation is
	# enemy-held VPs over the total in states you own, so handing away friendly
	# cities shrinks the denominator and makes every rebel gain count for more.
	# Without this Moskowien capitulates with Moskau still in its hands.
	# Removed in wif_rkm_civilwar.1 when the civil war is won.
	add_ideas = wif_rkm_civil_war_emergency

	# Whoever breaks away from Moskowien fights with Moskowien's technology.
	save_event_target_as = wif_tech_donor

	# Tell the world the Reichskommissariat has come apart.
	news_event = { id = wif_rkm_civilwar.2 }

	# The emergency governments form FIRST, so the rebels below have something
	# to declare war on. These are loyalist administrations, not rebellions -
	# the same thing happening in three cities rather than one.
	wif_nsp_formation = yes
	wif_nar_formation = yes
	wif_nwg_formation = yes

	# Novgorod: transferring a state is what brings NSR into existence.
	263 = { transfer_state_to = NSR }
	208 = { transfer_state_to = NSR }

	NSR = {
		# History files do not apply to countries created mid-game, so politics
		# and capital have to be set explicitly.
		set_capital = { state = 263 }

		set_politics = {
			ruling_party = communism
			last_election = "1936.1.1"
			election_frequency = 48
			elections_allowed = no
		}
		set_popularities = {
			democratic = 5
			fascism = 5
			communism = 90
			neutrality = 0
		}

		set_research_slots = 3
		set_stability = 0.6
		set_war_support = 0.8

		add_state_core = 263
		add_state_core = 208

		wif_inherit_tech_from_donor = yes

		add_ideas = wif_splinter_no_factions

		# Templates and divisions. load_oob rather than create_unit, which does
		# not fire reliably from this kind of context.
		load_oob = "NSR_revolt"

		# A thousand rifles of the best pattern Moskowien fields. The tech copy
		# above means infantry_equipment resolves to that same latest variant.
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 1000
			producer = NSR
		}

		declare_war_on = {
			target = RKM
			type = annex_everything
		}
		if = {
			limit = { country_exists = NSP }
			declare_war_on = {
				target = NSP
				type = annex_everything
			}
		}
		if = {
			limit = { country_exists = NAR }
			declare_war_on = {
				target = NAR
				type = annex_everything
			}
		}
		if = {
			limit = { country_exists = NWG }
			declare_war_on = {
				target = NWG
				type = annex_everything
			}
		}
	}

	# The Russian nationalists rise in the south-east at the same time, against
	# the Germans and the communists alike.
	wif_rlf_rising = yes

	# The labour camps of the west go up at the same time.
	wif_wrs_rising = yes

	# The two provisional governments proclaim themselves in the same days, one
	# in the south and one in the north. They form after the risings above so
	# each declares its wars once, in one direction - and the southern one first
	# so the northern one can see it exists and leave it alone.
	wif_srg_rising = yes
	wif_nrg_rising = yes

	# Lokot declares for Moskowien rather than against it, so it forms after the
	# risings it is meant to fight.
	wif_lok_formation = yes

	# If the Wehrmacht holds the balance of power when the rising breaks out, the
	# SS will not sit under an army-run administration while it burns. Tikhvin
	# goes with them, and they fight Moskowien and Novgorod both.
	if = {
		limit = {
			power_balance_value = {
				id = RKM_ss_wehrmacht_balance
				value > 0
			}
		}
		wif_ssm_mutiny = yes
	}

	# With the garrisons pulled west to fight the risings, the eastern marches
	# are simply walked into. Gorky, Kargopol and Kotlas go over to the Soviet
	# emergency government.
	wif_sme_eastern_annexation = yes
}

#

# well have been finished off before Moskowien ever comes apart. If it is gone,
# the three states simply stay where they are.

			add_state_core = 879
			add_state_core = 880
		}
	}
}

# The loyalist emergency government in Saint Petersburg. Run in RKM's scope.
# Formed before the rebels so they have something to declare war on.
wif_nsp_formation = {
	195 = { transfer_state_to = NSP }
	244 = { transfer_state_to = NSP }

	NSP = {
		set_capital = { state = 195 }

		set_politics = {
			ruling_party = fascism
			last_election = "1936.1.1"
			election_frequency = 48
			elections_allowed = no
		}
		set_popularities = {
			democratic = 5
			fascism = 85
			communism = 0
			neutrality = 10
		}

		set_research_slots = 2
		set_stability = 0.5
		set_war_support = 0.9

		# Only Saint Petersburg itself is theirs by right. Volkhov is held, not
		# claimed - it is Moskowien territory they are garrisoning. The Saint
		# Petersburg core passes to Moskowien on reunification.
		add_state_core = 195

		# Moskowien's technology, since Moskowien is who armed them.
		wif_inherit_tech_from_donor = yes

		add_ideas = wif_nsp_city_defence

		load_oob = "NSP_revolt"

		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 500
			producer = NSP
		}
	}

	# Bring them in on Moskowien's side. Follow whatever faction Moskowien is
	# actually in rather than assuming it is Germany's - only the leader of a
	# faction can admit new members, so find that leader by matching against
	# Moskowien's own membership.
	if = {
		limit = { is_faction_leader = yes }
		# Moskowien leads its own faction, so it admits them directly.
		add_to_faction = NSP
	}
	else_if = {
		limit = { is_in_faction = yes }
		every_country = {
			limit = {
				is_faction_leader = yes
				is_in_faction_with = PREV
			}
			add_to_faction = NSP
		}
	}

	# Fallback: Moskowien had no faction, or admission did not take. Mutual
	# guarantees at least bind them together. PREV rather than ROOT, since
	# inside the NSP scope PREV is unambiguously Moskowien.
	if = {
		limit = {
			NOT = { NSP = { is_in_faction_with = PREV } }
		}
		give_guarantee = NSP
		NSP = { give_guarantee = PREV }
	}

	# Locked only AFTER the faction work above, otherwise the rule would block
	# the very alliance it is meant to have. Its scripted place beside
	# Moskowien stands; it just cannot go looking for another.
	NSP = {
		add_ideas = wif_splinter_no_factions
	}
}

			last_election = "1936.1.1"
			election_frequency = 48
			elections_allowed = no
		}
		set_popularities = {
			democratic = 5
			fascism = 85
			communism = 0
			neutrality = 10
		}

		set_research_slots = 2
		set_stability = 0.5
		set_war_support = 0.9

		# Archangelsk is theirs by right. The core passes to Moskowien on
		# reunification.
		add_state_core = 214

		# Moskowien's technology, since Moskowien is who armed them. Saved as
		# wif_tech_donor at the top of wif_nsr_revolt, before the SS mutiny
		# reassigns it to Berlin.
		wif_inherit_tech_from_donor = yes

		add_ideas = wif_nar_city_defence

		load_oob = "NAR_revolt"

		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 500
			producer = NAR
		}
	}

	# Archangelsk stands alone. Deliberately NO faction and NO guarantees.
	#
	# This is the one emergency government that does not get Moskowien's
	# protection, and it is a design choice forced by the engine. The SME
	# declares on it the moment it forms, and that war is meant to be a private
	# duel - but HoI4 has no unjoinable war. A faction member who is attacked
	# drags the whole faction in automatically, and a guarantor gets a call to
	# arms it may well accept. The only way to keep a war between two countries
	# is for the defender to have nobody to call.
	#
	# So Archangelsk is a loyalist government that Moskowien cannot reach: too
	# far north, cut off by the risings, and left to hold the White Sea on its
	# own. Sankt Petersburg and Paulusberg keep their faction membership as
	# before - this applies to NAR only.
	NAR = {
		add_ideas = wif_splinter_no_factions
	}

	# The Soviet Military Emergency Government moves on Archangelsk at once. It
	# wants the northern ports and it will not get a better moment than the one
	# where Moskowien is coming apart everywhere else.
	if = {
		limit = { country_exists = SME }
		SME = {
			declare_war_on = {
				target = NAR
				type = annex_everything
			}
		}
	}
}

# The loyalist emergency government in Wolgograd. Run in RKM's scope.
# Formed before the rebels so they have something to declare war on.
#
# Identical in every respect to Sankt Petersburg's - see wif_nsp_formation for
# the reasoning behind the faction handling and the ordering. Anything changed
# there belongs here as well.
wif_nwg_formation = {
	217 = { transfer_state_to = NWG }

	NWG = {
		set_capital = { state = 217 }

		set_politics = {
			ruling_party = fascism
			last_election = "1936.1.1"
			election_frequency = 48
			elections_allowed = no
		}
		set_popularities = {
			democratic = 5
			fascism = 85
			communism = 0
			neutrality = 10
		}

		set_research_slots = 2
		set_stability = 0.5
		set_war_support = 0.9

		# Wolgograd is theirs by right. The core passes to Moskowien on
		# reunification.
		add_state_core = 217

		# Moskowien's technology, since Moskowien is who armed them. Saved as
		# wif_tech_donor at the top of wif_nsr_revolt, before the SS mutiny
		# reassigns it to Berlin.
		wif_inherit_tech_from_donor = yes

		add_ideas = wif_nwg_city_defence

		load_oob = "NWG_revolt"

		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 500
			producer = NWG
		}
	}

	# The government is named after the city it holds, and the city's name is not
	# fixed: Paulus Play renames state 217 to Paulusberg, SS Krieg leaves it as
	# Wolgograd. wif_rkm.21 sets the flag when the rename actually goes through.
	# Run in RKM's scope, so this reads Moskowien's flag.
	if = {
		limit = { has_country_flag = wif_paulusberg_renamed }
		NWG = { set_cosmetic_tag = NWG_paulusberg }
	}

	# Bring them in on Moskowien's side, following whatever faction Moskowien is
	# actually in rather than assuming it is Germany's.
	if = {
		limit = { is_faction_leader = yes }
		add_to_faction = NWG
	}
	else_if = {
		limit = { is_in_faction = yes }
		every_country = {
			limit = {
				is_faction_leader = yes
				is_in_faction_with = PREV
			}
			add_to_faction = NWG
		}
	}

	# Fallback: Moskowien had no faction, or admission did not take.
	if = {
		limit = {
			NOT = { NWG = { is_in_faction_with = PREV } }
		}
		give_guarantee = NWG
		NWG = { give_guarantee = PREV }
	}

	# Locked only AFTER the faction work above, otherwise the rule would block the
	# very alliance it is meant to have.
	NWG = {
		add_ideas = wif_splinter_no_factions
	}
}

# Ryazan changes hands. Called from on_state_control_changed, where ROOT is the
# new controller.
#
# Ryazan is where the labour camps are, and taking it is worth three militia
# divisions of freed prisoners to whoever does it - but only to the risen
# states. Moskowien is the gaoler, Sankt Petersburg and Lokot are fighting for
# it, and the SS mutiny is not liberating anyone, so the eligible tags are named
# outright rather than carved out with exceptions.
#
# The flag is per country, so each of them collects once and the state cannot be
# traded back and forth to farm divisions.
wif_ryazan_liberation_check = {
	if = {
		limit = {
			OR = {
				tag = NSR
				tag = RLF
				tag = WRS
				tag = SRG
				tag = NRG
			}
			controls_state = 254
			NOT = { has_country_flag = wif_ryazan_liberated }
		}

		set_country_flag = wif_ryazan_liberated
		load_oob = "wif_ryazan_liberation"
	}
}
