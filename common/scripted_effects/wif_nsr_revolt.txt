# The Novgorod rising.
#
# Deliberately NOT start_civil_war: that would split Moskowien's army between
# the two sides. Moskowien keeps every division it has, and the rebels get
# purpose-built formations of their own from NSR_revolt.txt instead.
#
# Run in RKM's scope.
wif_nsr_revolt = {
	# Marks the civil war as running. Cleared when Moskowien has put it down,
	# which is what triggers Saint Petersburg folding back in.
	set_country_flag = wif_moskowien_civil_war

	# Stamp every state Moskowien holds right now, BEFORE a single one is
	# handed to a breakaway. If Moskowien wins, everything carrying this flag
	# comes back, whoever is sitting on it by then - which stops Germany or an
	# opportunist neighbour from pocketing the wreckage of the civil war.
	every_owned_state = {
		set_state_flag = wif_prewar_rkm_state
	}

	# Whoever breaks away from Moskowien fights with Moskowien's technology.
	save_event_target_as = wif_tech_donor

	# Tell the world the Reichskommissariat has come apart.
	news_event = { id = wif_rkm_civilwar.2 }

	# Saint Petersburg forms FIRST, so the rebels below have something to
	# declare war on. It is a loyalist emergency government, not a rebellion.
	wif_nsp_formation = yes

	# Novgorod: transferring a state is what brings NSR into existence.
	263 = { transfer_state_to = NSR }
	208 = { transfer_state_to = NSR }

	NSR = {
		# History files do not apply to countries created mid-game, so politics
		# and capital have to be set explicitly.
		set_capital = { state = 263 }

		set_politics = {
			ruling_party = communism
			last_election = "1936.1.1"
			election_frequency = 48
			elections_allowed = no
		}
		set_popularities = {
			democratic = 5
			fascism = 5
			communism = 90
			neutrality = 0
		}

		set_research_slots = 3
		set_stability = 0.6
		set_war_support = 0.8

		add_state_core = 263
		add_state_core = 208

		wif_inherit_tech_from_donor = yes

		add_ideas = wif_splinter_no_factions

		# Templates and divisions. load_oob rather than create_unit, which does
		# not fire reliably from this kind of context.
		load_oob = "NSR_revolt"

		# A thousand rifles of the best pattern Moskowien fields. The tech copy
		# above means infantry_equipment resolves to that same latest variant.
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 1000
			producer = NSR
		}

		declare_war_on = {
			target = RKM
			type = annex_everything
		}
		if = {
			limit = { country_exists = NSP }
			declare_war_on = {
				target = NSP
				type = annex_everything
			}
		}
	}

	# The Russian nationalists rise in the south-east at the same time, against
	# the Germans and the communists alike.
	wif_rlf_rising = yes

	# The labour camps of the west go up at the same time.
	wif_wrs_rising = yes

	# Lokot declares for Moskowien rather than against it, so it forms after the
	# risings it is meant to fight.
	wif_lok_formation = yes

	# If the Wehrmacht holds the balance of power when the rising breaks out, the
	# SS will not sit under an army-run administration while it burns. Tikhvin
	# goes with them, and they fight Moskowien and Novgorod both.
	if = {
		limit = {
			power_balance_value = {
				id = RKM_ss_wehrmacht_balance
				value > 0
			}
		}
		wif_ssm_mutiny = yes
	}
}

# The Lokot Autonomy. Loyalist, like Saint Petersburg: it forms on Moskowien's
# side and folds back in if Moskowien wins. Run in RKM's scope.
wif_lok_formation = {
	224 = { transfer_state_to = LOK }

	LOK = {
		set_capital = { state = 224 }

		set_politics = {
			ruling_party = fascism
			last_election = "1936.1.1"
			election_frequency = 48
			elections_allowed = no
		}
		set_popularities = {
			democratic = 5
			fascism = 85
			communism = 0
			neutrality = 10
		}

		set_research_slots = 2
		set_stability = 0.5
		set_war_support = 0.85

		# Kept, and passed to Moskowien on reintegration.
		add_state_core = 224

		wif_inherit_tech_from_donor = yes

		load_oob = "LOK_revolt"

		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 300
			producer = LOK
		}

		# Into the fight from the first day rather than waiting to be dragged in
		# by the faction. Every rising that already exists, which is all of them
		# except the SS - Lokot forms before that one, and the SS declare on
		# Lokot themselves.
		if = {
			limit = { country_exists = NSR }
			declare_war_on = {
				target = NSR
				type = annex_everything
			}
		}
		if = {
			limit = { country_exists = RLF }
			declare_war_on = {
				target = RLF
				type = annex_everything
			}
		}
		if = {
			limit = { country_exists = WRS }
			declare_war_on = {
				target = WRS
				type = annex_everything
			}
		}
	}

	# Same faction handling as Saint Petersburg: follow Moskowien's own
	# membership rather than assuming Germany leads it.
	if = {
		limit = { is_faction_leader = yes }
		add_to_faction = LOK
	}
	else_if = {
		limit = { is_in_faction = yes }
		every_country = {
			limit = {
				is_faction_leader = yes
				is_in_faction_with = PREV
			}
			add_to_faction = LOK
		}
	}

	if = {
		limit = {
			NOT = { LOK = { is_in_faction_with = PREV } }
		}
		give_guarantee = LOK
		LOK = { give_guarantee = PREV }
	}

	# Locked only after the faction work, or the rule would block the alliance
	# it exists to have.
	LOK = {
		add_ideas = wif_splinter_no_factions
	}
}

# The loyalist emergency government in Saint Petersburg. Run in RKM's scope.
# Formed before the rebels so they have something to declare war on.
wif_nsp_formation = {
	195 = { transfer_state_to = NSP }
	244 = { transfer_state_to = NSP }

	NSP = {
		set_capital = { state = 195 }

		set_politics = {
			ruling_party = fascism
			last_election = "1936.1.1"
			election_frequency = 48
			elections_allowed = no
		}
		set_popularities = {
			democratic = 5
			fascism = 85
			communism = 0
			neutrality = 10
		}

		set_research_slots = 2
		set_stability = 0.5
		set_war_support = 0.9

		# Only Saint Petersburg itself is theirs by right. Volkhov is held, not
		# claimed - it is Moskowien territory they are garrisoning. The Saint
		# Petersburg core passes to Moskowien on reunification.
		add_state_core = 195

		# Moskowien's technology, since Moskowien is who armed them.
		wif_inherit_tech_from_donor = yes

		add_ideas = wif_nsp_city_defence

		load_oob = "NSP_revolt"

		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 500
			producer = NSP
		}
	}

	# Bring them in on Moskowien's side. Follow whatever faction Moskowien is
	# actually in rather than assuming it is Germany's - only the leader of a
	# faction can admit new members, so find that leader by matching against
	# Moskowien's own membership.
	if = {
		limit = { is_faction_leader = yes }
		# Moskowien leads its own faction, so it admits them directly.
		add_to_faction = NSP
	}
	else_if = {
		limit = { is_in_faction = yes }
		every_country = {
			limit = {
				is_faction_leader = yes
				is_in_faction_with = PREV
			}
			add_to_faction = NSP
		}
	}

	# Fallback: Moskowien had no faction, or admission did not take. Mutual
	# guarantees at least bind them together. PREV rather than ROOT, since
	# inside the NSP scope PREV is unambiguously Moskowien.
	if = {
		limit = {
			NOT = { NSP = { is_in_faction_with = PREV } }
		}
		give_guarantee = NSP
		NSP = { give_guarantee = PREV }
	}

	# Locked only AFTER the faction work above, otherwise the rule would block
	# the very alliance it is meant to have. Its scripted place beside
	# Moskowien stands; it just cannot go looking for another.
	NSP = {
		add_ideas = wif_splinter_no_factions
	}
}
