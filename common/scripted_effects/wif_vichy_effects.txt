# VIC (Vichy) scope. Called by events/France.txt (france.14, every "Agree"
# option) the moment Vichy France is set up: Vichy starts as a German
# satellite (vanilla autonomy_satellite) instead of a guaranteed state or an
# ordinary puppet.
wif_vichy_become_satellite = {
	if = {
		limit = {
			country_exists = GER
			NOT = { is_subject_of = GER }
		}
		GER = { puppet = PREV }
	}
	if = {
		limit = { is_subject_of = GER }
		GER = {
			set_autonomy = {
				target = PREV
				autonomous_state = autonomy_satellite
				end_wars = no
				end_civil_wars = no
			}
		}
	}
}

# VIC scope. Sovereignty Restored: no longer a German subject, but still in the
# Axis. The faction leader is read before the subject relationship ends, and
# takes Vichy back at once if ending it took Vichy out of the faction.
wif_vichy_sovereignty = {
	if = {
		limit = { is_in_faction = yes }
		random_country = {
			limit = {
				is_faction_leader = yes
				is_in_faction_with = ROOT
			}
			save_event_target_as = wif_vic_axis_leader
		}
	}
	if = {
		limit = { is_subject = yes }
		OVERLORD = { end_puppet = ROOT }
	}
	if = {
		limit = {
			is_in_faction = no
			has_event_target = wif_vic_axis_leader
		}
		event_target:wif_vic_axis_leader = { add_to_faction = ROOT }
	}
	set_country_flag = wif_vic_sovereign
}

# VIC scope. The Axis Is Our Shield.
wif_vichy_axis_shield = {
	if = {
		limit = { is_in_faction = no }
		# a German power leading a faction first, else the leader of any
		# faction that has a German power in it
		if = {
			limit = {
				any_country = {
					is_faction_leader = yes
					wif_cw_is_german_power = yes
				}
			}
			random_country = {
				limit = {
					is_faction_leader = yes
					wif_cw_is_german_power = yes
				}
				add_to_faction = ROOT
			}
		}
		else = {
			random_country = {
				limit = {
					is_faction_leader = yes
					any_country = {
						wif_cw_is_german_power = yes
						is_in_faction_with = PREV
					}
				}
				add_to_faction = ROOT
			}
		}
	}
	every_country = {
		limit = { is_in_faction_with = ROOT NOT = { tag = ROOT } }
		add_opinion_modifier = { target = ROOT modifier = wif_vic_axis_shield }
	}
}
