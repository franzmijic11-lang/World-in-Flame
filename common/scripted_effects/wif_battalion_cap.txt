# Battalion soft cap.
#
#     CAP = 324 * ( 1 + SUM of every percentage below )
#
# Every bonus is a percentage OF THE 324 BASE, and they are SUMMED. Nothing
# compounds: a +50% conscription law and a +70% war support bonus give
# 324 * 2.2, not 324 * 1.5 * 1.7. Keep it that way - the whole design rests on
# the player being able to add the percentages up in their head.
#
# THE SUM CAN GO NEGATIVE. The economy law band runs from -50% under
# Undisturbed Isolation to +25% under Totaler Krieg, so a quiet isolationist
# country sits below the base rather than merely failing to climb above it.
#
# THERE IS NO DESIGN FLOOR. There used to be one at the base itself, on the
# grounds that nothing could be negative; both that assumption and the clamp
# are gone. All that remains at the bottom of wif_recalculate_battalion_cap is
# a one-battalion guard, there so the headroom division cannot fail.
#
# TWO SOURCES ARE FLAT rather than percentages - territory and the single-state
# penalty, both counted in battalions and both applied after the multiply.
#
# Only LINE battalions count. Support companies and regiments are excluded and
# are neither counted against the cap nor helped by it.
#
# This is a soft cap. Going over is always allowed; it just costs you. The
# penalties live in common/dynamic_modifiers/wif_battalion_cap.txt.
#
#   Territory       MEASURED IN BATTALIONS, not percent, and added after the
#                   multiply. Per owned state: cored with a city province 2,
#                   plus 0.5 for each further city province in that state;
#                   cored without one 0.5; not cored 0.1. Deliberately blind
#                   to how many people live in the state.
#   Russian warlord 80%, for the successor states of the Soviet collapse that
#                   hold their territory by the army and nothing else.
#   Red Army        216 BATTALIONS, flat, while the Red Army national spirit
#                   is held; 432 once The Glory of the Red Army is completed.
#                   Also applied after the multiply.
#   Chinese millions  A MULTIPLIER, not a bonus: x2 on the finished cap, for
#                   China and Communist China. Applied after everything else.
#   Single state    a country owning exactly one state is pulled back to a
#                   flat 27 battalions before other modifiers. Also in
#                   battalions, also applied after the multiply.
#   Conscription    disarmed 0, volunteer 10%, limited 15%, extensive 25%,
#                   service by requirement 35%, all adults serve 45%,
#                   scraping the barrel 50%.
#   Economy law     THE ONLY NEGATIVE SOURCE. Undisturbed isolation -50%,
#                   isolation -25%, civilian economy -15%, early mobilisation
#                   0, partial mobilisation 10%, war economy 20%, total
#                   mobilisation 15%, totaler krieg 25%, national defence
#                   state 20%, new economic policy 10%, capital investment
#                   model 5%, reichskommissariat economy 20%.
#   Standing        major power 864 BATTALIONS flat, faction leader 50%.
#                   These stack, but in different units - see the standing
#                   block. Both are re-read weekly so they follow the
#                   country's actual status rather than being granted once.
#   Stability       0-20 nothing, 21-40 5%, 41-60 7%, 61-80 9%, 81-100 10%.
#   War support     0-20 nothing, 21-40 10%, 41-60 30%, 61-80 50%, 81-100 70%.
#   Wartime         additive, up to +185%, ALL of it stripped the moment the
#                   country is at peace again. The headline clause is a
#                   three-way read on the balance of strength - see below.
#   World tension   highest qualifying band only, never summed with itself.
#
# No flat bonuses anywhere yet. If one is ever added it goes on after the
# multiply, not into wif_bat_pct.


# Territory component. THE ONLY PART OF THE CAP MEASURED IN BATTALIONS RATHER
# THAN IN PERCENT, and the only part that has to walk every state, so it is
# cached in wif_terr_bat and recomputed ONLY when the country's territory
# actually changes - see common/on_actions/wif_on_actions.txt. Running this
# loop weekly for every country in the game is what would make the cap
# expensive, and there is no reason to: borders do not move most weeks.
#
# It is flat because it is now specified in battalions - a cored city state is
# worth two battalions, not a fraction of the base. Flat sources are added
# AFTER the percentage multiply, which is exactly where the header of this
# file always said they would have to go.
#
#   cored, has a city province      2 battalions, plus 0.5 for every further
#                                   city province in the same state
#   cored, no city province         1 battalion, or 0.5 under 150k people
#   not cored, whatever it holds    0.1 battalions
#
# A CITY PROVINCE IS A PROVINCE WITH urban TERRAIN, counted off the map by
# tools/city_provinces.py and seeded into the state variable wif_city_provs by
# common/scripted_effects/wif_city_provinces.txt. It has to be precomputed:
# HoI4 has no province scope, no province iterator, and has_terrain is country
# scope only, so a state cannot be asked what its provinces are made of.
#
# Note this is NOT has_state_category any more. The two disagree on 160 states
# - 106 have a city province under a plain category, 54 have a city category
# with no city province on the map - and the province is the thing the player
# can actually see.
wif_recalculate_territory_bonus = {
	set_variable = { wif_terr_bat = 0 }

	every_owned_state = {
		if = {
			limit = { is_core_of = PREV }

			if = {
				limit = { check_variable = { wif_city_provs > 0 } }
				add_to_variable = { PREV.wif_terr_bat = 2 }

				# Every city province after the first is worth another half a
				# battalion. Read into a temp so the subtraction cannot touch
				# the seeded state variable itself - that value is set once at
				# startup and must survive the whole game.
				set_temp_variable = { wif_tmp_extra_cities = wif_city_provs }
				subtract_from_temp_variable = { wif_tmp_extra_cities = 1 }
				multiply_temp_variable = { wif_tmp_extra_cities = 0.5 }
				add_to_variable = { PREV.wif_terr_bat = wif_tmp_extra_cities }
			}
			else = {
				# A cored state with no city on it is worth a whole battalion,
				# unless there is nobody in it: under 150,000 people it pays
				# the half battalion this tier used to pay throughout.
				#
				# state_population_k is the population in thousands, read into
				# a temp first the way vanilla reads it in AST.txt - and the
				# way num_owned_states is read further down - rather than
				# compared inside check_variable directly.
				set_temp_variable = { wif_tmp_pop_k = state_population_k }
				if = {
					limit = { check_variable = { wif_tmp_pop_k < 150 } }
					add_to_variable = { PREV.wif_terr_bat = 0.5 }
				}
				else = {
					add_to_variable = { PREV.wif_terr_bat = 1 }
				}
			}
		}
		else = {
			# Occupied ground pays a token amount whatever is standing on it -
			# a captured metropolis is worth the same as captured tundra until
			# it is cored.
			add_to_variable = { PREV.wif_terr_bat = 0.1 }
		}
	}
}


# The weekly pass. Variable maths only - no loops, no unit iteration.
wif_recalculate_battalion_cap = {
	# Refresh the cached territory bonus only when the country's territory has
	# actually moved. num_owned_states and num_core_states are plain country
	# variables, so this is two integer compares in the common case and the
	# every_owned_state loop runs only on the weeks a border or a core changed.
	# That is what keeps a per-country weekly pass affordable.
	#
	# The one thing it will not catch on its own is a state changing category
	# (a town growing into a city) with no change to either count. That is rare
	# and self-corrects the next time either number moves.
	#
	# The has_variable check also seeds countries that formed mid-game and so
	# never saw on_startup.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = wif_terr_bat }
				NOT = { check_variable = { wif_terr_states = num_owned_states } }
				NOT = { check_variable = { wif_terr_cores = num_core_states } }
			}
		}
		wif_recalculate_territory_bonus = yes
		set_variable = { wif_terr_states = num_owned_states }
		set_variable = { wif_terr_cores = num_core_states }
	}

	# Seed the limit at the bare base before anything else touches it. This is
	# a diagnostic as much as a safety net: the readout previously showed a
	# limit of 0, which can only happen if the variable is never written, so
	# seeding it here means the counter can never show 0 again. If it ever
	# reads exactly 324 on a country that should have bonuses, the block at
	# the bottom of this effect is not being reached.
	set_variable = { wif_bat_limit = 324 }

	# Every bonus below is banked in its own wif_src_* variable instead of being
	# added straight into a running total, so the officer corps tooltip can show
	# the player exactly which bonuses are live and what each one is worth. They
	# are summed into wif_bat_pct in one place, further down.
	#
	# All of them are zeroed every pass. That is what makes the breakdown honest:
	# a bonus that stopped applying this week reads 0 rather than keeping last
	# week's value, and the tooltip hides any line sitting at 0.
	set_variable = { wif_src_consc = 0 }
	set_variable = { wif_src_econ = 0 }
	set_variable = { wif_src_warlord = 0 }
	set_variable = { wif_src_redarmy = 0 }
	set_variable = { wif_src_stab = 0 }
	set_variable = { wif_src_ws = 0 }
	set_variable = { wif_src_major = 0 }
	set_variable = { wif_src_faction = 0 }
	set_variable = { wif_src_war_bal = 0 }
	set_variable = { wif_src_war_def = 0 }
	set_variable = { wif_src_war_2x = 0 }
	set_variable = { wif_src_war_4x = 0 }
	set_variable = { wif_src_tension = 0 }

	# The inputs themselves, copied so the tooltip can state the reason a band
	# was picked ("Stability 72%") next to what it paid out. Read here rather
	# than in localisation so the tooltip can never disagree with the maths.
	set_variable = { wif_val_stab = has_stability }
	set_variable = { wif_val_ws = has_war_support }
	set_variable = { wif_val_tension = threat }

	# --- Conscription law -------------------------------------------------
	# Disarmed nation is deliberately absent: it contributes nothing, so a
	# disarmed country sits on the bare 324 plus whatever else it has earned.
	# These read the vanilla law ideas directly, so the recruitable population
	# factors on the laws themselves are untouched.
	if = {
		limit = { has_idea = scraping_the_barrel }
		set_variable = { wif_src_consc = 0.50 }
	}
	else_if = {
		limit = { has_idea = all_adults_serve }
		set_variable = { wif_src_consc = 0.45 }
	}
	else_if = {
		limit = { has_idea = service_by_requirement }
		set_variable = { wif_src_consc = 0.35 }
	}
	else_if = {
		limit = { has_idea = extensive_conscription }
		set_variable = { wif_src_consc = 0.25 }
	}
	else_if = {
		limit = { has_idea = limited_conscription }
		set_variable = { wif_src_consc = 0.15 }
	}
	else_if = {
		limit = { has_idea = volunteer_only }
		set_variable = { wif_src_consc = 0.10 }
	}

	# --- Economy law ------------------------------------------------------
	# THE ONLY SOURCE THAT CAN GO NEGATIVE. An isolationist economy does not
	# just fail to help the army, it holds it down: the plant, the transport
	# and the political will to keep men in uniform are not there to be had.
	#
	# Early Mobilisation is deliberately absent from this chain, exactly as
	# Disarmed Nation is absent from the conscription chain above. It is the
	# neutral point at 0%, wif_src_econ is already zeroed, and a source worth
	# nothing is hidden from the tooltip rather than shown as "+0%".
	#
	# TOTAL ECONOMIC MOBILISATION PAYS LESS THAN WAR ECONOMY, 15% against 20%,
	# and that is not a typo. Total mobilisation is the economy taking the
	# manpower; the army sees less of it, not more.
	#
	# Eleven of the twelve laws are vanilla, four of them normally reachable by
	# one country only - totaler_krieg_economy (GER), national_defense_state
	# (JAP), new_economic_policy (SOV) and capital_investment_model. The
	# twelfth, wif_reichskommissariat_economy, is this mod's own and lives in
	# common/ideas/wif_economy_laws.txt.
	#
	# A country holds exactly one economy law, so the order of these branches
	# is presentation only. They run highest to lowest to match the way the
	# header lists them.
	if = {
		limit = { has_idea = totaler_krieg_economy }
		set_variable = { wif_src_econ = 0.25 }
	}
	else_if = {
		limit = { has_idea = war_economy }
		set_variable = { wif_src_econ = 0.20 }
	}
	else_if = {
		limit = { has_idea = national_defense_state }
		set_variable = { wif_src_econ = 0.20 }
	}
	else_if = {
		limit = { has_idea = wif_reichskommissariat_economy }
		set_variable = { wif_src_econ = 0.20 }
	}
	else_if = {
		limit = { has_idea = tot_economic_mobilisation }
		set_variable = { wif_src_econ = 0.15 }
	}
	else_if = {
		limit = { has_idea = partial_economic_mobilisation }
		set_variable = { wif_src_econ = 0.10 }
	}
	else_if = {
		limit = { has_idea = new_economic_policy }
		set_variable = { wif_src_econ = 0.10 }
	}
	else_if = {
		limit = { has_idea = capital_investment_model }
		set_variable = { wif_src_econ = 0.05 }
	}
	else_if = {
		limit = { has_idea = civilian_economy }
		set_variable = { wif_src_econ = -0.15 }
	}
	else_if = {
		limit = { has_idea = isolation }
		set_variable = { wif_src_econ = -0.25 }
	}
	else_if = {
		limit = { has_idea = undisturbed_isolation }
		set_variable = { wif_src_econ = -0.50 }
	}

	# --- Russian warlord ---------------------------------------------------
	# The successor states hold their ground with the army they inherited and
	# nothing else - no recognised borders, no allies, no economy that answers
	# to them. What they do have is every man who used to serve the Union, and
	# no government left to tell him to go home.
	#
	# Read off the idea rather than the tag list so a country that somehow ends
	# up with the spirit gets the cap that goes with it, and so this stays one
	# line no matter how many successors the collapse eventually produces. The
	# grant is in common/on_actions/wif_on_actions.txt, gated on the trigger
	# wif_is_russian_warlord.
	if = {
		limit = { has_idea = wif_russian_warlord }
		set_variable = { wif_src_warlord = 0.80 }
	}

	# --- The Red Army ------------------------------------------------------
	# A flat 216 battalions while the country carries the Red Army national
	# spirit, rising to 432 once it has completed The Glory of the Red Army.
	# Two thirds of the base, then four thirds of it.
	#
	# THE SPIRIT IS A DYNAMIC MODIFIER, not an idea. Vanilla builds the Red
	# Army spirit as SOV_the_red_army_dynamic_modifier, whose every line reads
	# a SOV_the_red_army_* variable that focuses add to - which is how one
	# spirit accumulates a whole branch of bonuses without ever being swapped
	# out. The fascist and neutrality path carries the _alt copy of it instead,
	# and that is the "any spirit it turns into" case: both are read here.
	#
	# The focus has an _alt twin for the same reason, gated by allow_branch on
	# government, so both spellings count.
	#
	# Read off the spirit rather than off tag = SOV deliberately. If a
	# successor state ever inherits the Red Army spirit it inherits the cap
	# that goes with it, which is the whole point of the spirit meaning
	# something.
	set_variable = { wif_src_redarmy = 0 }
	if = {
		limit = {
			OR = {
				has_dynamic_modifier = SOV_the_red_army_dynamic_modifier
				has_dynamic_modifier = SOV_the_red_army_dynamic_modifier_alt
			}
		}
		set_variable = { wif_src_redarmy = 216 }

		if = {
			limit = {
				OR = {
					has_completed_focus = SOV_the_glory_of_the_red_army
					has_completed_focus = SOV_the_glory_of_the_red_army_alt
				}
			}
			set_variable = { wif_src_redarmy = 432 }
		}
	}

	# --- Single state ------------------------------------------------------
	# A country whose legal territory is one single state is not a country that
	# can field an army, and the base of 324 says otherwise, so it is taken
	# away again.
	#
	# The target is a flat 27 battalions before any other modifier - a twelfth
	# of the base. 324 - 297 = 27, hence the -297, and the state's own
	# territory worth is then subtracted as well so that the answer is exactly
	# 27 whether that one state is a cored capital with three cities or a
	# scrap of occupied ground. The penalty is defined as "whatever it takes to
	# land on 27", not as a number that happens to be near it.
	#
	# Percentage bonuses still work normally on top: a one-state country at
	# +100% reaches 324 * 2 - 297 = 351. It is the base that is taken away,
	# not the country's ability to earn.
	#
	# Measured in BATTALIONS, so it is applied after the multiply with the
	# territory. num_owned_states is the legal count, which is what was asked
	# for - occupied ground the country does not own does not save it.
	# Read through a temp rather than testing num_owned_states inside
	# check_variable directly. Copying a game variable into one of ours is the
	# form this file already proves works - it is how wif_terr_states is set -
	# and a comparison the parser refuses would fail silently, taking the rest
	# of the effect with it.
	set_variable = { wif_src_onestate = 0 }
	set_temp_variable = { wif_tmp_owned_states = num_owned_states }
	if = {
		limit = { check_variable = { wif_tmp_owned_states < 2 } }
		set_variable = { wif_src_onestate = -297 }
		subtract_from_variable = { wif_src_onestate = wif_terr_bat }
	}

	# --- Stability --------------------------------------------------------
	if = {
		limit = { has_stability > 0.8 }
		set_variable = { wif_src_stab = 0.10 }
	}
	else_if = {
		limit = { has_stability > 0.6 }
		set_variable = { wif_src_stab = 0.09 }
	}
	else_if = {
		limit = { has_stability > 0.4 }
		set_variable = { wif_src_stab = 0.07 }
	}
	else_if = {
		limit = { has_stability > 0.2 }
		set_variable = { wif_src_stab = 0.05 }
	}

	# --- War support ------------------------------------------------------
	if = {
		limit = { has_war_support > 0.8 }
		set_variable = { wif_src_ws = 0.70 }
	}
	else_if = {
		limit = { has_war_support > 0.6 }
		set_variable = { wif_src_ws = 0.50 }
	}
	else_if = {
		limit = { has_war_support > 0.4 }
		set_variable = { wif_src_ws = 0.30 }
	}
	else_if = {
		limit = { has_war_support > 0.2 }
		set_variable = { wif_src_ws = 0.10 }
	}

	# --- Standing ----------------------------------------------------------
	# Both are re-read from scratch every weekly pass, which is what makes them
	# self-correcting: a minor that becomes a major picks the bonus up on the
	# next tick, and a major knocked down to minor loses it just as
	# automatically. Nothing is granted or revoked by event, and there is no
	# flag to leave stale.
	#
	# THE TWO ARE MEASURED IN DIFFERENT UNITS. Major power is 864 BATTALIONS,
	# flat, applied after the multiply with territory and the Red Army. Faction
	# leader is still a percentage of the base. So they still stack, but not
	# into one number: a major that also leads a faction gets 864 battalions
	# and +50%.
	#
	# Flat is the point. As a percentage, being a major was worth 324
	# battalions and grew with every other bonus the country had; as a flat
	# 864 it is worth the same to a major that has just been knocked about as
	# to one at the height of its powers, and it is worth far more than it was.
	if = {
		limit = { is_major = yes }
		set_variable = { wif_src_major = 864 }
	}
	if = {
		limit = { is_faction_leader = yes }
		set_variable = { wif_src_faction = 0.50 }
	}

	# --- Line battalions in service ---------------------------------------
	# Computed here rather than down with the cap because the wartime clauses
	# below weigh this country's strength against its enemies', and both sides
	# of that comparison have to be line-only for it to mean anything.
	#
	# num_battalions counts EVERY sub unit in a division, support companies
	# included, so the support total is subtracted back out.
	wif_count_support_companies = yes
	set_variable = { wif_bat_used = num_battalions }
	subtract_from_variable = { wif_bat_used = wif_support_bat }
	if = {
		limit = { check_variable = { wif_bat_used < 0 } }
		set_variable = { wif_bat_used = 0 }
	}

	# --- Wartime ----------------------------------------------------------
	# Everything in this block is conditional on has_war = yes, which is what
	# makes it evaporate at peace: the next weekly pass after the war ends
	# simply does not add it, the cap drops, and the country has a month of
	# grace to demobilise into the smaller number.
	#
	# Enemy strength is summed across every country currently at war with this
	# one, which is what makes the alliance clause work without a separate
	# faction lookup - if the enemy is in a faction, every member that joined
	# the war is already in the totals.
	set_variable = { wif_enemy_bat = 0 }
	set_variable = { wif_enemy_mil = 0 }
	if = {
		limit = { has_war = yes }

		# Each enemy has its own support companies stripped out the same way,
		# so this is line battalions measured against line battalions.
		every_enemy_country = {
			wif_count_support_companies = yes
			set_temp_variable = { wif_enemy_line = num_battalions }
			subtract_from_temp_variable = { wif_enemy_line = wif_support_bat }
			add_to_variable = { PREV.wif_enemy_bat = wif_enemy_line }
			add_to_variable = { PREV.wif_enemy_mil = num_of_military_factories }
		}

		# --- Balance of strength -------------------------------------------
		# The headline wartime clause. It used to be a single flat +100% for
		# "enemy no larger than you", judged on battalions alone, which paid a
		# small industrial power the same as a giant and rewarded exactly the
		# countries that needed it least.
		#
		# Strength is now a WEIGHTED RATIO, 40% battalions and 60% military
		# factories:
		#
		#     ratio = 0.4 * (enemy battalions / yours)
		#           + 0.6 * (enemy mil factories / yours)
		#
		# Ratios rather than raw totals, deliberately. Adding battalions and
		# factories together directly would be meaningless - a major fields
		# thousands of the first and hundreds of the second, so the factory
		# term would vanish into rounding and the 60/40 split would not be the
		# split it claims to be. Comparing each dimension against your own and
		# THEN weighting keeps both terms dimensionless and keeps the weights
		# honest.
		#
		# Factories carry more weight than battalions because an army in the
		# field is what you have now, while an arms industry is what you will
		# still be fielding in a year. A country out-produced two to one is in
		# more trouble than one merely outnumbered, and the cap should give it
		# the room to answer.
		#
		#     ratio < 0.9    enemy is weaker      +20%
		#     0.9 - 1.1      evenly matched       +25%
		#     ratio > 1.1    enemy is stronger    +60%
		#
		# Beating someone smaller than you is no longer a reason to be handed a
		# vast army allowance; being outmatched is.
		#
		# Both denominators are floored at 1 so a country with no army, or none
		# of its own arms industry, cannot divide by zero.
		set_temp_variable = { wif_own_bat_safe = wif_bat_used }
		if = {
			limit = { check_variable = { wif_own_bat_safe < 1 } }
			set_temp_variable = { wif_own_bat_safe = 1 }
		}

		set_temp_variable = { wif_own_mil_safe = num_of_military_factories }
		if = {
			limit = { check_variable = { wif_own_mil_safe < 1 } }
			set_temp_variable = { wif_own_mil_safe = 1 }
		}

		set_temp_variable = { wif_ratio_bat = wif_enemy_bat }
		divide_temp_variable = { wif_ratio_bat = wif_own_bat_safe }
		multiply_temp_variable = { wif_ratio_bat = 0.4 }

		set_temp_variable = { wif_ratio_mil = wif_enemy_mil }
		divide_temp_variable = { wif_ratio_mil = wif_own_mil_safe }
		multiply_temp_variable = { wif_ratio_mil = 0.6 }

		# Kept as a country variable rather than a temp one so the officer
		# corps tooltip can show the player the figure the band was chosen on.
		set_variable = { wif_val_enemy_ratio = wif_ratio_bat }
		add_to_variable = { wif_val_enemy_ratio = wif_ratio_mil }

		# check_variable has no <= or >= operator, only > < = - which is why
		# the even band is the else rather than a pair of bounds. Using <=
		# here once broke the parse of this whole file.
		if = {
			limit = { check_variable = { wif_val_enemy_ratio < 0.9 } }
			set_variable = { wif_src_war_bal = 0.20 }
		}
		else_if = {
			limit = { check_variable = { wif_val_enemy_ratio > 1.1 } }
			set_variable = { wif_src_war_bal = 0.60 }
		}
		else = {
			set_variable = { wif_src_war_bal = 0.25 }
		}

		if = {
			limit = { has_defensive_war = yes }
			set_variable = { wif_src_war_def = 0.15 }
		}

		# Outnumbered two to one, and four to one. These stack with each
		# other by design: a country facing 4x gets both, +110% between them.
		set_variable = { wif_enemy_bat_x2 = wif_bat_used }
		multiply_variable = { wif_enemy_bat_x2 = 2 }
		set_variable = { wif_enemy_bat_x4 = wif_bat_used }
		multiply_variable = { wif_enemy_bat_x4 = 4 }

		if = {
			limit = { check_variable = { wif_enemy_bat > wif_enemy_bat_x2 } }
			set_variable = { wif_src_war_2x = 0.20 }
		}
		if = {
			# Same restriction as above: no >= operator, so "at least four times"
			# is written as "NOT less than four times".
			limit = { NOT = { check_variable = { wif_enemy_bat < wif_enemy_bat_x4 } } }
			set_variable = { wif_src_war_4x = 0.90 }
		}
	}

	# --- World tension ----------------------------------------------------
	# Highest band only. Written as if/else_if precisely so 80%+ tension gives
	# 60% and not 30+40+60.
	if = {
		limit = { threat > 0.7999 }
		set_variable = { wif_src_tension = 0.60 }
	}
	else_if = {
		limit = { threat > 0.5999 }
		set_variable = { wif_src_tension = 0.40 }
	}
	else_if = {
		limit = { threat > 0.1999 }
		set_variable = { wif_src_tension = 0.30 }
	}

	# --- Total -------------------------------------------------------------
	# The one place every bonus is combined. Territory is already a single
	# cached variable of its own; the rest come from the wif_src_* set above.
	# Nothing else may add to wif_bat_pct - a bonus that skips this block would
	# be invisible in the tooltip breakdown, which is exactly the kind of quiet
	# disagreement between display and maths this structure exists to prevent.
	# Territory is NOT here any more. It is measured in battalions and applied
	# after the multiply, together with the single-state penalty.
	set_variable = { wif_bat_pct = 0 }
	add_to_variable = { wif_bat_pct = wif_src_consc }
	add_to_variable = { wif_bat_pct = wif_src_econ }
	add_to_variable = { wif_bat_pct = wif_src_warlord }
	add_to_variable = { wif_bat_pct = wif_src_stab }
	add_to_variable = { wif_bat_pct = wif_src_ws }
	add_to_variable = { wif_bat_pct = wif_src_faction }
	add_to_variable = { wif_bat_pct = wif_src_war_bal }
	add_to_variable = { wif_bat_pct = wif_src_war_def }
	add_to_variable = { wif_bat_pct = wif_src_war_2x }
	add_to_variable = { wif_bat_pct = wif_src_war_4x }
	add_to_variable = { wif_bat_pct = wif_src_tension }

	# --- Cap ---------------------------------------------------------------
	# Deliberately NOT rounded here. round_variable sat between wif_bat_used
	# and this variable, and wif_bat_limit was the only thing downstream of it
	# that failed to reach the interface while wif_bat_used arrived fine - the
	# readout showed the battalion count but no limit. The display formatter
	# in the localisation ("|0") already rounds to whole battalions, so the
	# rounding was never needed for presentation, and leaving the fractional
	# value in the variable does not affect any comparison below.
	set_variable = { wif_bat_limit = 1 }
	add_to_variable = { wif_bat_limit = wif_bat_pct }
	multiply_variable = { wif_bat_limit = 324 }

	# --- Flat battalions ---------------------------------------------------
	# Everything above is a percentage of the base. Everything here is a count
	# of battalions, added after the multiply so that percentage bonuses never
	# scale it - two battalions for a city state is two battalions whether the
	# country is Luxembourg or the Reich.
	add_to_variable = { wif_bat_limit = wif_src_major }
	add_to_variable = { wif_bat_limit = wif_terr_bat }
	add_to_variable = { wif_bat_limit = wif_src_redarmy }
	add_to_variable = { wif_bat_limit = wif_src_onestate }

	# --- The multiplier ----------------------------------------------------
	# Applied LAST, to the finished figure, which is what makes it a doubling
	# rather than another bonus. It is deliberately not a wif_src_* variable:
	# everything with that prefix is additive and is either zeroed and summed
	# into wif_bat_pct or added to wif_bat_limit, and a multiplier that joined
	# either of those sets would be wrong in a way the tooltip could not show.
	#
	# Reset to 1 rather than 0 every pass, for the obvious reason.
	#
	# One edge case, and it is harmless: a China reduced to a single state has
	# its cap pinned to 27 by the flat penalty above, and this then doubles it
	# to 54. A country in that position is not being held back by its cap.
	set_variable = { wif_cap_multiplier = 1 }
	if = {
		limit = { has_idea = wif_chinese_millions }
		set_variable = { wif_cap_multiplier = 2 }
	}
	multiply_variable = { wif_bat_limit = wif_cap_multiplier }

	# THERE IS NO LONGER A DESIGN FLOOR. The cap used to be clamped up to the
	# base, then to half of it; both swallowed penalties that are meant to
	# land, so both are gone. A country that has earned a small cap gets a
	# small cap.
	#
	# What remains is a guard, not a floor. The headroom fraction below divides
	# by this variable, and the AI strategy thresholds read that fraction, so a
	# zero or negative cap would either fail the division or silently invert
	# every threshold the AI reacts to. One battalion is the smallest value
	# that keeps both honest.
	#
	# It is reachable: a single-state country on Undisturbed Isolation totals
	# well below zero, and it is supposed to.
	if = {
		limit = { check_variable = { wif_bat_limit < 1 } }
		set_variable = { wif_bat_limit = 1 }
	}

	# Spare allowance, read by the AI strategies in
	# common/ai_strategy/wif_battalion_cap.txt.
	#
	# Expressed as a FRACTION of the country's own cap as well as a raw count,
	# for the same reason the factory cap uses a ratio: a flat "20 battalions
	# left" means nothing in common between Luxembourg and Germany, so the
	# thresholds the AI reacts to have to be proportional. wif_bat_limit is
	# guarded at a minimum of 1 above, so this can never divide by zero.
	set_variable = { wif_bat_headroom = wif_bat_limit }
	subtract_from_variable = { wif_bat_headroom = wif_bat_used }

	set_variable = { wif_bat_headroom_pct = wif_bat_headroom }
	divide_variable = { wif_bat_headroom_pct = wif_bat_limit }

	# wif_bat_used was worked out above, before the wartime comparisons.
	set_variable = { wif_battalions_over = wif_bat_used }
	subtract_from_variable = { wif_battalions_over = wif_bat_limit }
	if = {
		limit = { check_variable = { wif_battalions_over < 0 } }
		set_variable = { wif_battalions_over = 0 }
	}

	# Per-battalion penalty rates. These are the numbers to edit to rebalance
	# the whole system; the dynamic modifier just reads the results.
	#
	# HoI4 reads these as fractions, so 0.02 is +2% attrition PER BATTALION
	# over. Ten battalions over costs +20% attrition, +20% supply consumption,
	# -10% political power gain and -1% factory output.
	#
	# Political power used to be the outlier at -0.1, ten times the rest, which
	# zeroed political power gain at only ten battalions over. It is now -0.01:
	# -1% per battalion over, in line with attrition and supply. Gain now
	# reaches zero at a hundred over rather than ten.
	#
	# The overshoot is still NOT clamped. A country 150 over runs at -150% and
	# bleeds political power rather than merely stopping earning it. If it
	# should stop at zero instead, clamp wif_over_pp at -1 below.
	set_variable = { wif_over_attrition = wif_battalions_over }
	multiply_variable = { wif_over_attrition = 0.02 }

	set_variable = { wif_over_supply = wif_battalions_over }
	multiply_variable = { wif_over_supply = 0.02 }

	set_variable = { wif_over_pp = wif_battalions_over }
	multiply_variable = { wif_over_pp = -0.01 }

	set_variable = { wif_over_factory = wif_battalions_over }
	multiply_variable = { wif_over_factory = -0.001 }

	# --- Grace period ------------------------------------------------------
	# One month to demobilise. The clock starts the week a country first goes
	# over its cap, which covers the case the grace exists for: a war ends, the
	# wartime bonuses vanish, and an army that was legal last week is suddenly
	# 200 battalions over. Dip back under and the latch clears, so the next
	# overshoot gets a fresh month rather than an already-expired one.
	if = {
		limit = { check_variable = { wif_battalions_over > 0 } }

		if = {
			limit = { NOT = { has_country_flag = wif_over_cap_latch } }
			set_country_flag = wif_over_cap_latch
			set_country_flag = { flag = wif_bat_grace days = 30 value = 1 }
		}

		if = {
			limit = {
				NOT = { has_country_flag = wif_bat_grace }
				NOT = { has_dynamic_modifier = { modifier = wif_over_battalion_cap } }
			}
			add_dynamic_modifier = { modifier = wif_over_battalion_cap }
		}
	}
	else = {
		clr_country_flag = wif_over_cap_latch
		clr_country_flag = wif_bat_grace

		if = {
			limit = { has_dynamic_modifier = { modifier = wif_over_battalion_cap } }
			remove_dynamic_modifier = { modifier = wif_over_battalion_cap }
		}
	}
}

# --- AI demobilisation -------------------------------------------------------
#
# Makes the AI actually shed divisions when it is over the cap, rather than
# only slowing recruitment and sitting there bleeding penalties forever. The
# ai_strategy file next door stops it building MORE; this is what brings an
# army that is already over back down.
#
# AI ONLY. A player who chooses to sit over the cap and eat the penalties is
# making a decision, and having divisions vanish out from under them without
# asking would be indefensible. The AI cannot make that decision for itself.
#
# IT ONLY EVER SHEDS THE OVERSHOOT. That is the whole safety property of this
# system, so it is worth being exact about how it is guaranteed:
#
#   wif_disband_allowance = ( wif_battalions_over / average battalions per
#                             division ) * 0.75
#
# That is how many DIVISIONS the country is over by, and it is the hard budget
# for the entire pass. Clearing a state subtracts exactly the number of
# divisions that state held, and no state is ever cleared unless its whole
# garrison fits inside what is left. When the allowance runs out the pass
# stops. An army ten divisions too big loses ten divisions, not an eleventh,
# and an army at its cap loses nothing at all. It happens fast, in one weekly
# pass, bounded only by the second guarantee below: no pass may take more
# than half the divisions the country actually has.
#
# The 0.75 is a deliberate margin. The allowance is derived from the AVERAGE
# division size, but the divisions actually disbanded may each be larger than
# average, which would shed more battalions than intended. Undershooting is
# harmless - the weekly pass runs again in seven days and takes another bite -
# whereas overshooting is precisely the failure this margin exists to prevent.
#
# WHY delete_unit AND NOT destroy_unit. The engine gives two ways to remove a
# division and they trade against each other:
#
#   destroy_unit  - division scope, takes exactly the division you scoped, but
#                   there is NO refund. The equipment and the men are gone.
#   delete_unit   - REFUNDS equipment and manpower with disband = yes, exactly
#                   as the player's own disband button does, but it is COUNTRY
#                   scope and its finest aim is a template or a state.
#
# No third effect is both precise and refunding. The refund is the entire point
# of a demobilisation - a country standing its army down should get its rifles
# and its men back - so this uses delete_unit and accepts that the unit of work
# is a STATE: every division standing in a chosen state is disbanded together,
# which is exactly why the ladder below must know how many that is before it
# commits to anything.
#
# THE LADDER. divisions_in_state can only compare against a literal, so a
# state's garrison cannot be read into a variable and weighed against the
# allowance directly. wif_disband_state_if_it_fits brackets it instead: branch
# N matches a state holding exactly N divisions, checks that the allowance
# covers N, and subtracts N when it clears it. States holding more than 10
# divisions are left alone entirely - at that size it is an army group rather
# than a garrison, and breaking one up is not demobilisation.
#
# REAR FIRST. Divisions come from states with no enemy-controlled neighbour
# before anything near the fighting is touched. The rear/front test is
# vanilla's own, from CHI_decisions.txt: a state is on the front if any
# neighbouring state is controlled by somebody at war with us. At peace nothing
# is at war with us, so every state reads as rear and the first pass covers the
# whole country - correct, since a country at peace has no front to hold.
#
# The scope shape is vanilla's too, from Finland's crack_down_on_the_military
# focus: iterate controlled states, test divisions_in_state from country scope
# with state = PREV, then step back up with PREV = { delete_unit = { state =
# PREV ... } }. PREV resolves back down to the state from inside the country
# block. Neither if nor a scripted effect call changes scope, so the chain
# survives both the helper call and the ladder.
#
# Nothing here re-reads num_divisions or wif_bat_used mid-loop. Those do not
# refresh inside a single effect, which is what made the old factory
# enforcement pass eat every arms factory in the country. The allowance is
# computed once, up front, and from then on only ever decremented.
wif_disband_over_cap = {
	if = {
		limit = {
			is_ai = yes
			check_variable = { wif_battalions_over > 0 }
			NOT = { has_country_flag = wif_bat_grace }
			check_variable = { num_divisions > 0 }
		}

		# The country, reachable from inside the state loop. The neighbour test
		# below is two scopes deep and an event target reads far better there
		# than a PREV chain.
		save_event_target_as = wif_disband_country

		# Average line battalions per division, so an overshoot measured in
		# battalions can be turned into a number of divisions. Floored at 1 so
		# the division below cannot blow up.
		set_variable = { wif_disband_avg = wif_bat_used }
		divide_variable = { wif_disband_avg = num_divisions }
		if = {
			limit = { check_variable = { wif_disband_avg < 1 } }
			set_variable = { wif_disband_avg = 1 }
		}

		set_variable = { wif_disband_allowance = wif_battalions_over }
		divide_variable = { wif_disband_allowance = wif_disband_avg }
		multiply_variable = { wif_disband_allowance = 0.75 }

		# THE ONE THING THIS MUST NEVER DO IS EMPTY AN ARMY.
		#
		# Speed is wanted here - a country that has overbuilt should shed it in a
		# week or two, not trickle down over a season - so the overshoot budget
		# above is deliberately NOT rate limited. It is bounded instead, and the
		# bound is a share of the army rather than a flat number of divisions, so
		# it scales with whoever is being corrected.
		#
		# In the normal case this changes nothing: a country 10 divisions over with
		# 100 in the field is bounded by the overshoot, sheds all 10 in one pass,
		# and is back at its cap the same week.
		#
		# WHAT IT GUARDS. wif_disband_allowance is a count of DIVISIONS derived by
		# dividing a count of BATTALIONS by the average division size. That average
		# is floored at 1, and if it ever bottoms out - wif_bat_used reading low or
		# stale against a live num_divisions, exactly the kind of desync the
		# check_variable parse bug once caused - the allowance stops being "the
		# divisions we are over by" and becomes "the battalions we are over by,
		# spent as whole divisions". A country 100 battalions over would be handed
		# an allowance of 75 divisions and disband every unit it has.
		#
		# Half the army per pass makes that unreachable. The overshoot bound stays
		# the operative limit in every healthy case; this is purely the floor under
		# a bad one.
		set_variable = { wif_disband_max = num_divisions }
		multiply_variable = { wif_disband_max = 0.5 }
		clamp_variable = {
			var = wif_disband_allowance
			min = 0
			max = wif_disband_max
		}

		# Pass 1 - the rear. Almost always the only pass that runs, and the only
		# one that can run at all in peacetime.
		every_controlled_state = {
			limit = {
				NOT = {
					any_neighbor_state = {
						controller = { has_war_with = event_target:wif_disband_country }
					}
				}
			}
			wif_disband_state_if_it_fits = yes
		}

		# Pass 2 - the front, and only for a country more than 20% over, the same
		# threshold wif_bat_far_over_cap uses. Reached only when the rear is
		# already dry and allowance is still left, which means the country is deep
		# enough over that the attrition and supply bill costs it more than the
		# divisions are worth.
		#
		# This can only ever fire during a war: at peace every state reads as rear
		# and pass 1 will have spent the allowance first. Deleting this block
		# leaves a rear-only system that still converges once a war ends and every
		# state becomes rear again.
		if = {
			limit = { check_variable = { wif_bat_headroom_pct < -0.2 } }
			every_controlled_state = {
				wif_disband_state_if_it_fits = yes
			}
		}
	}
}


# Clears one state's garrison, but only if that whole garrison fits inside what
# is left of wif_disband_allowance. Called in STATE scope from the loops above,
# so THIS is the state and PREV is the country.
#
# Branch N matches a state holding EXACTLY N divisions - more than N-1, and not
# more than N - which is what makes the subtraction exact. Both bounds are
# needed: without the lower one a small garrison would fall through into a
# larger branch and over-subtract the allowance.
#
# check_variable has no <= or >= operator, only > < =, so "the allowance covers
# N" has to be written as NOT = { allowance < N }. Do not "simplify" it.
wif_disband_state_if_it_fits = {
	if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 0 }
				NOT = { divisions_in_state = { state = PREV size > 1 } }
				NOT = { check_variable = { wif_disband_allowance < 1 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 1 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 1 }
				NOT = { divisions_in_state = { state = PREV size > 2 } }
				NOT = { check_variable = { wif_disband_allowance < 2 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 2 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 2 }
				NOT = { divisions_in_state = { state = PREV size > 3 } }
				NOT = { check_variable = { wif_disband_allowance < 3 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 3 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 3 }
				NOT = { divisions_in_state = { state = PREV size > 4 } }
				NOT = { check_variable = { wif_disband_allowance < 4 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 4 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 4 }
				NOT = { divisions_in_state = { state = PREV size > 5 } }
				NOT = { check_variable = { wif_disband_allowance < 5 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 5 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 5 }
				NOT = { divisions_in_state = { state = PREV size > 6 } }
				NOT = { check_variable = { wif_disband_allowance < 6 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 6 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 6 }
				NOT = { divisions_in_state = { state = PREV size > 7 } }
				NOT = { check_variable = { wif_disband_allowance < 7 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 7 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 7 }
				NOT = { divisions_in_state = { state = PREV size > 8 } }
				NOT = { check_variable = { wif_disband_allowance < 8 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 8 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 8 }
				NOT = { divisions_in_state = { state = PREV size > 9 } }
				NOT = { check_variable = { wif_disband_allowance < 9 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 9 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 9 }
				NOT = { divisions_in_state = { state = PREV size > 10 } }
				NOT = { check_variable = { wif_disband_allowance < 10 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 10 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
}

# NOTE: the "one civilian dies per soldier killed" population drain has now
# been proposed and dropped twice. It is technically buildable - casualties is
# a readable cumulative variable, so weekly deaths are a delta against last
# week's reading, and add_manpower works in STATE scope - but it is
# deliberately NOT part of this system. Do not add it back without asking.


# Total support companies currently fielded.
#
# num_battalions counts EVERY sub unit in every division, support companies
# included - support companies register under num_battalions_with_type@ just
# like line battalions do. So the line-only figure the cap is built around has
# to subtract them back out, one sub unit type at a time, because there is no
# aggregate "all support" variable.
#
# This list is every vanilla sub unit whose type block contains "support",
# generated from common/units/ rather than typed by hand. A mod that adds a
# NEW support company will not be in here and will be miscounted as a line
# battalion until its name is added below.
wif_count_support_companies = {
