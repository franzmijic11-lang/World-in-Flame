# Battalion soft cap.
#
#     CAP = 432 * ( 1 + SUM of every percentage below )
#
# Every bonus is a percentage OF THE 432 BASE, and they are SUMMED. Nothing
# compounds: a +50% conscription law and a +70% war support bonus give
# 432 * 2.2, not 432 * 1.5 * 1.7. Keep it that way - the whole design rests on
# the player being able to add the percentages up in their head.
#
# 432 is a hard floor. Since no component is ever negative the sum cannot drag
# the cap below the base, but the clamp is written out anyway so that adding a
# malus component later cannot silently break the floor.
#
# Only LINE battalions count. Support companies and regiments are excluded and
# are neither counted against the cap nor helped by it.
#
# This is a soft cap. Going over is always allowed; it just costs you. The
# penalties live in common/dynamic_modifiers/wif_battalion_cap.txt.
#
#   Territory       per owned state: cored 0.2%, cored with a city 0.3% total,
#                   uncored 0.05%. Linear, and deliberately blind to how many
#                   people live in the state.
#   Conscription    disarmed 0, volunteer 10%, limited 15%, extensive 25%,
#                   service by requirement 35%, all adults serve 45%,
#                   scraping the barrel 50%.
#   Standing        major power 100%, faction leader 50%. These stack, and
#                   both are re-read weekly so they follow the country's
#                   actual status rather than being granted once.
#   Stability       0-20 nothing, 21-40 5%, 41-60 7%, 61-80 9%, 81-100 10%.
#   War support     0-20 nothing, 21-40 10%, 41-60 30%, 61-80 50%, 81-100 70%.
#   Wartime         additive, up to +225%, ALL of it stripped the moment the
#                   country is at peace again.
#   World tension   highest qualifying band only, never summed with itself.
#
# No flat bonuses anywhere yet. If one is ever added it goes on after the
# multiply, not into wif_bat_pct.


# Territory component. This is the only part that needs to walk every state, so
# it is cached in wif_terr_pct and recomputed ONLY when the country's territory
# actually changes - see common/on_actions/wif_on_actions.txt. Running this
# loop weekly for every country in the game is what would make the cap
# expensive, and there is no reason to: borders do not move most weeks.
wif_recalculate_territory_bonus = {
	set_variable = { wif_terr_pct = 0 }

	every_owned_state = {
		if = {
			limit = { is_core_of = PREV }
			add_to_variable = { PREV.wif_terr_pct = 0.002 }

			# A city is worth an extra 0.1% on top of the 0.2%, so a cored
			# state with one is 0.3% in total rather than 0.2% + 0.3%.
			# "Has a city" is read off the state category, which is the only
			# thing in the game that actually tracks urbanisation.
			if = {
				limit = {
					OR = {
						has_state_category = city
						has_state_category = large_city
						has_state_category = metropolis
						has_state_category = megalopolis
					}
				}
				add_to_variable = { PREV.wif_terr_pct = 0.001 }
			}
		}
		else = {
			# Occupied ground pays a quarter of what cored ground pays.
			add_to_variable = { PREV.wif_terr_pct = 0.0005 }
		}
	}
}


# The weekly pass. Variable maths only - no loops, no unit iteration.
wif_recalculate_battalion_cap = {
	# Refresh the cached territory bonus only when the country's territory has
	# actually moved. num_owned_states and num_core_states are plain country
	# variables, so this is two integer compares in the common case and the
	# every_owned_state loop runs only on the weeks a border or a core changed.
	# That is what keeps a per-country weekly pass affordable.
	#
	# The one thing it will not catch on its own is a state changing category
	# (a town growing into a city) with no change to either count. That is rare
	# and self-corrects the next time either number moves.
	#
	# The has_variable check also seeds countries that formed mid-game and so
	# never saw on_startup.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = wif_terr_pct }
				NOT = { check_variable = { wif_terr_states = num_owned_states } }
				NOT = { check_variable = { wif_terr_cores = num_core_states } }
			}
		}
		wif_recalculate_territory_bonus = yes
		set_variable = { wif_terr_states = num_owned_states }
		set_variable = { wif_terr_cores = num_core_states }
	}

	# Seed the limit at the bare base before anything else touches it. This is
	# a diagnostic as much as a safety net: the readout previously showed a
	# limit of 0, which can only happen if the variable is never written, so
	# seeding it here means the counter can never show 0 again. If it ever
	# reads exactly 432 on a country that should have bonuses, the block at
	# the bottom of this effect is not being reached.
	set_variable = { wif_bat_limit = 432 }

	# Every bonus below is banked in its own wif_src_* variable instead of being
	# added straight into a running total, so the officer corps tooltip can show
	# the player exactly which bonuses are live and what each one is worth. They
	# are summed into wif_bat_pct in one place, further down.
	#
	# All of them are zeroed every pass. That is what makes the breakdown honest:
	# a bonus that stopped applying this week reads 0 rather than keeping last
	# week's value, and the tooltip hides any line sitting at 0.
	set_variable = { wif_src_consc = 0 }
	set_variable = { wif_src_stab = 0 }
	set_variable = { wif_src_ws = 0 }
	set_variable = { wif_src_major = 0 }
	set_variable = { wif_src_faction = 0 }
	set_variable = { wif_src_war_even = 0 }
	set_variable = { wif_src_war_def = 0 }
	set_variable = { wif_src_war_2x = 0 }
	set_variable = { wif_src_war_4x = 0 }
	set_variable = { wif_src_tension = 0 }

	# The inputs themselves, copied so the tooltip can state the reason a band
	# was picked ("Stability 72%") next to what it paid out. Read here rather
	# than in localisation so the tooltip can never disagree with the maths.
	set_variable = { wif_val_stab = has_stability }
	set_variable = { wif_val_ws = has_war_support }
	set_variable = { wif_val_tension = threat }

	# --- Conscription law -------------------------------------------------
	# Disarmed nation is deliberately absent: it contributes nothing, so a
	# disarmed country sits on the bare 432 plus whatever else it has earned.
	# These read the vanilla law ideas directly, so the recruitable population
	# factors on the laws themselves are untouched.
	if = {
		limit = { has_idea = scraping_the_barrel }
		set_variable = { wif_src_consc = 0.50 }
	}
	else_if = {
		limit = { has_idea = all_adults_serve }
		set_variable = { wif_src_consc = 0.45 }
	}
	else_if = {
		limit = { has_idea = service_by_requirement }
		set_variable = { wif_src_consc = 0.35 }
	}
	else_if = {
		limit = { has_idea = extensive_conscription }
		set_variable = { wif_src_consc = 0.25 }
	}
	else_if = {
		limit = { has_idea = limited_conscription }
		set_variable = { wif_src_consc = 0.15 }
	}
	else_if = {
		limit = { has_idea = volunteer_only }
		set_variable = { wif_src_consc = 0.10 }
	}

	# --- Stability --------------------------------------------------------
	if = {
		limit = { has_stability > 0.8 }
		set_variable = { wif_src_stab = 0.10 }
	}
	else_if = {
		limit = { has_stability > 0.6 }
		set_variable = { wif_src_stab = 0.09 }
	}
	else_if = {
		limit = { has_stability > 0.4 }
		set_variable = { wif_src_stab = 0.07 }
	}
	else_if = {
		limit = { has_stability > 0.2 }
		set_variable = { wif_src_stab = 0.05 }
	}

	# --- War support ------------------------------------------------------
	if = {
		limit = { has_war_support > 0.8 }
		set_variable = { wif_src_ws = 0.70 }
	}
	else_if = {
		limit = { has_war_support > 0.6 }
		set_variable = { wif_src_ws = 0.50 }
	}
	else_if = {
		limit = { has_war_support > 0.4 }
		set_variable = { wif_src_ws = 0.30 }
	}
	else_if = {
		limit = { has_war_support > 0.2 }
		set_variable = { wif_src_ws = 0.10 }
	}

	# --- Standing ----------------------------------------------------------
	# Both of these are re-read from scratch every weekly pass, which is what
	# makes them self-correcting: a minor that becomes a major picks up the
	# +100% on the next tick, and a major that is knocked down to minor loses
	# it just as automatically. Nothing needs to be granted or revoked by
	# event, and there is no flag to leave stale.
	#
	# They stack. A major that also leads a faction gets both, +150%.
	if = {
		limit = { is_major = yes }
		set_variable = { wif_src_major = 1.00 }
	}
	if = {
		limit = { is_faction_leader = yes }
		set_variable = { wif_src_faction = 0.50 }
	}

	# --- Line battalions in service ---------------------------------------
	# Computed here rather than down with the cap because the wartime clauses
	# below weigh this country's strength against its enemies', and both sides
	# of that comparison have to be line-only for it to mean anything.
	#
	# num_battalions counts EVERY sub unit in a division, support companies
	# included, so the support total is subtracted back out.
	wif_count_support_companies = yes
	set_variable = { wif_bat_used = num_battalions }
	subtract_from_variable = { wif_bat_used = wif_support_bat }
	if = {
		limit = { check_variable = { wif_bat_used < 0 } }
		set_variable = { wif_bat_used = 0 }
	}

	# --- Wartime ----------------------------------------------------------
	# Everything in this block is conditional on has_war = yes, which is what
	# makes it evaporate at peace: the next weekly pass after the war ends
	# simply does not add it, the cap drops, and the country has a month of
	# grace to demobilise into the smaller number.
	#
	# Enemy strength is measured as the total line battalions of every country
	# currently at war with this one, which is what makes the alliance clause
	# work without a separate faction lookup - if the enemy is in a faction,
	# every member that joined the war is already in this sum.
	set_variable = { wif_enemy_bat = 0 }
	if = {
		limit = { has_war = yes }

		# Each enemy has its own support companies stripped out the same way,
		# so this is line battalions measured against line battalions.
		every_enemy_country = {
			wif_count_support_companies = yes
			set_temp_variable = { wif_enemy_line = num_battalions }
			subtract_from_temp_variable = { wif_enemy_line = wif_support_bat }
			add_to_variable = { PREV.wif_enemy_bat = wif_enemy_line }
		}

		# Fighting someone your own size or smaller. The big one, and the
		# reason a defensive war against a peer is survivable at all.
		if = {
			# check_variable has no <= operator - only > < = - so "enemy is the
			# same size or smaller" has to be written as "NOT bigger". Using <=
			# here broke the parse of this whole file.
			limit = { NOT = { check_variable = { wif_enemy_bat > wif_bat_used } } }
			set_variable = { wif_src_war_even = 1.00 }
		}

		if = {
			limit = { has_defensive_war = yes }
			set_variable = { wif_src_war_def = 0.15 }
		}

		# Outnumbered two to one, and four to one. These stack with each
		# other by design: a country facing 4x gets both, +110% between them.
		set_variable = { wif_enemy_bat_x2 = wif_bat_used }
		multiply_variable = { wif_enemy_bat_x2 = 2 }
		set_variable = { wif_enemy_bat_x4 = wif_bat_used }
		multiply_variable = { wif_enemy_bat_x4 = 4 }

		if = {
			limit = { check_variable = { wif_enemy_bat > wif_enemy_bat_x2 } }
			set_variable = { wif_src_war_2x = 0.20 }
		}
		if = {
			# Same restriction as above: no >= operator, so "at least four times"
			# is written as "NOT less than four times".
			limit = { NOT = { check_variable = { wif_enemy_bat < wif_enemy_bat_x4 } } }
			set_variable = { wif_src_war_4x = 0.90 }
		}
	}

	# --- World tension ----------------------------------------------------
	# Highest band only. Written as if/else_if precisely so 80%+ tension gives
	# 60% and not 30+40+60.
	if = {
		limit = { threat > 0.7999 }
		set_variable = { wif_src_tension = 0.60 }
	}
	else_if = {
		limit = { threat > 0.5999 }
		set_variable = { wif_src_tension = 0.40 }
	}
	else_if = {
		limit = { threat > 0.1999 }
		set_variable = { wif_src_tension = 0.30 }
	}

	# --- Total -------------------------------------------------------------
	# The one place every bonus is combined. Territory is already a single
	# cached variable of its own; the rest come from the wif_src_* set above.
	# Nothing else may add to wif_bat_pct - a bonus that skips this block would
	# be invisible in the tooltip breakdown, which is exactly the kind of quiet
	# disagreement between display and maths this structure exists to prevent.
	set_variable = { wif_bat_pct = wif_terr_pct }
	add_to_variable = { wif_bat_pct = wif_src_consc }
	add_to_variable = { wif_bat_pct = wif_src_stab }
	add_to_variable = { wif_bat_pct = wif_src_ws }
	add_to_variable = { wif_bat_pct = wif_src_major }
	add_to_variable = { wif_bat_pct = wif_src_faction }
	add_to_variable = { wif_bat_pct = wif_src_war_even }
	add_to_variable = { wif_bat_pct = wif_src_war_def }
	add_to_variable = { wif_bat_pct = wif_src_war_2x }
	add_to_variable = { wif_bat_pct = wif_src_war_4x }
	add_to_variable = { wif_bat_pct = wif_src_tension }

	# --- Cap ---------------------------------------------------------------
	# Deliberately NOT rounded here. round_variable sat between wif_bat_used
	# and this variable, and wif_bat_limit was the only thing downstream of it
	# that failed to reach the interface while wif_bat_used arrived fine - the
	# readout showed the battalion count but no limit. The display formatter
	# in the localisation ("|0") already rounds to whole battalions, so the
	# rounding was never needed for presentation, and leaving the fractional
	# value in the variable does not affect any comparison below.
	set_variable = { wif_bat_limit = 1 }
	add_to_variable = { wif_bat_limit = wif_bat_pct }
	multiply_variable = { wif_bat_limit = 432 }

	# 432 is the floor, always.
	if = {
		limit = { check_variable = { wif_bat_limit < 432 } }
		set_variable = { wif_bat_limit = 432 }
	}

	# Spare allowance, read by the AI strategies in
	# common/ai_strategy/wif_battalion_cap.txt.
	#
	# Expressed as a FRACTION of the country's own cap as well as a raw count,
	# for the same reason the factory cap uses a ratio: a flat "20 battalions
	# left" means nothing in common between Luxembourg and Germany, so the
	# thresholds the AI reacts to have to be proportional. wif_bat_limit is
	# floored at 432 above, so this can never divide by zero.
	set_variable = { wif_bat_headroom = wif_bat_limit }
	subtract_from_variable = { wif_bat_headroom = wif_bat_used }

	set_variable = { wif_bat_headroom_pct = wif_bat_headroom }
	divide_variable = { wif_bat_headroom_pct = wif_bat_limit }

	# wif_bat_used was worked out above, before the wartime comparisons.
	set_variable = { wif_battalions_over = wif_bat_used }
	subtract_from_variable = { wif_battalions_over = wif_bat_limit }
	if = {
		limit = { check_variable = { wif_battalions_over < 0 } }
		set_variable = { wif_battalions_over = 0 }
	}

	# Per-battalion penalty rates. These are the numbers to edit to rebalance
	# the whole system; the dynamic modifier just reads the results.
	#
	# HoI4 reads these as fractions, so 0.02 is +2% attrition PER BATTALION
	# over. Ten battalions over costs +20% attrition, +20% supply consumption,
	# -10% political power gain and -1% factory output.
	#
	# Political power used to be the outlier at -0.1, ten times the rest, which
	# zeroed political power gain at only ten battalions over. It is now -0.01:
	# -1% per battalion over, in line with attrition and supply. Gain now
	# reaches zero at a hundred over rather than ten.
	#
	# The overshoot is still NOT clamped. A country 150 over runs at -150% and
	# bleeds political power rather than merely stopping earning it. If it
	# should stop at zero instead, clamp wif_over_pp at -1 below.
	set_variable = { wif_over_attrition = wif_battalions_over }
	multiply_variable = { wif_over_attrition = 0.02 }

	set_variable = { wif_over_supply = wif_battalions_over }
	multiply_variable = { wif_over_supply = 0.02 }

	set_variable = { wif_over_pp = wif_battalions_over }
	multiply_variable = { wif_over_pp = -0.01 }

	set_variable = { wif_over_factory = wif_battalions_over }
	multiply_variable = { wif_over_factory = -0.001 }

	# --- Grace period ------------------------------------------------------
	# One month to demobilise. The clock starts the week a country first goes
	# over its cap, which covers the case the grace exists for: a war ends, the
	# wartime bonuses vanish, and an army that was legal last week is suddenly
	# 200 battalions over. Dip back under and the latch clears, so the next
	# overshoot gets a fresh month rather than an already-expired one.
	if = {
		limit = { check_variable = { wif_battalions_over > 0 } }

		if = {
			limit = { NOT = { has_country_flag = wif_over_cap_latch } }
			set_country_flag = wif_over_cap_latch
			set_country_flag = { flag = wif_bat_grace days = 30 value = 1 }
		}

		if = {
			limit = {
				NOT = { has_country_flag = wif_bat_grace }
				NOT = { has_dynamic_modifier = { modifier = wif_over_battalion_cap } }
			}
			add_dynamic_modifier = { modifier = wif_over_battalion_cap }
		}
	}
	else = {
		clr_country_flag = wif_over_cap_latch
		clr_country_flag = wif_bat_grace

		if = {
			limit = { has_dynamic_modifier = { modifier = wif_over_battalion_cap } }
			remove_dynamic_modifier = { modifier = wif_over_battalion_cap }
		}
	}
}

# --- AI demobilisation -------------------------------------------------------
#
# Makes the AI actually shed divisions when it is over the cap, rather than
# only slowing recruitment and sitting there bleeding penalties forever. The
# ai_strategy file next door stops it building MORE; this is what brings an
# army that is already over back down.
#
# AI ONLY. A player who chooses to sit over the cap and eat the penalties is
# making a decision, and having divisions vanish out from under them without
# asking would be indefensible. The AI cannot make that decision for itself.
#
# IT ONLY EVER SHEDS THE OVERSHOOT. That is the whole safety property of this
# system, so it is worth being exact about how it is guaranteed:
#
#   wif_disband_allowance = ( wif_battalions_over / average battalions per
#                             division ) * 0.75
#
# That is how many DIVISIONS the country is over by, and it is the hard budget
# for the entire pass. Clearing a state subtracts exactly the number of
# divisions that state held, and no state is ever cleared unless its whole
# garrison fits inside what is left. When the allowance runs out the pass
# stops. An army ten divisions too big loses ten divisions, not an eleventh,
# and an army at its cap loses nothing at all.
#
# The 0.75 is a deliberate margin. The allowance is derived from the AVERAGE
# division size, but the divisions actually disbanded may each be larger than
# average, which would shed more battalions than intended. Undershooting is
# harmless - the weekly pass runs again in seven days and takes another bite -
# whereas overshooting is precisely the failure this margin exists to prevent.
#
# WHY delete_unit AND NOT destroy_unit. The engine gives two ways to remove a
# division and they trade against each other:
#
#   destroy_unit  - division scope, takes exactly the division you scoped, but
#                   there is NO refund. The equipment and the men are gone.
#   delete_unit   - REFUNDS equipment and manpower with disband = yes, exactly
#                   as the player's own disband button does, but it is COUNTRY
#                   scope and its finest aim is a template or a state.
#
# No third effect is both precise and refunding. The refund is the entire point
# of a demobilisation - a country standing its army down should get its rifles
# and its men back - so this uses delete_unit and accepts that the unit of work
# is a STATE: every division standing in a chosen state is disbanded together,
# which is exactly why the ladder below must know how many that is before it
# commits to anything.
#
# THE LADDER. divisions_in_state can only compare against a literal, so a
# state's garrison cannot be read into a variable and weighed against the
# allowance directly. wif_disband_state_if_it_fits brackets it instead: branch
# N matches a state holding exactly N divisions, checks that the allowance
# covers N, and subtracts N when it clears it. States holding more than 10
# divisions are left alone entirely - at that size it is an army group rather
# than a garrison, and breaking one up is not demobilisation.
#
# REAR FIRST. Divisions come from states with no enemy-controlled neighbour
# before anything near the fighting is touched. The rear/front test is
# vanilla's own, from CHI_decisions.txt: a state is on the front if any
# neighbouring state is controlled by somebody at war with us. At peace nothing
# is at war with us, so every state reads as rear and the first pass covers the
# whole country - correct, since a country at peace has no front to hold.
#
# The scope shape is vanilla's too, from Finland's crack_down_on_the_military
# focus: iterate controlled states, test divisions_in_state from country scope
# with state = PREV, then step back up with PREV = { delete_unit = { state =
# PREV ... } }. PREV resolves back down to the state from inside the country
# block. Neither if nor a scripted effect call changes scope, so the chain
# survives both the helper call and the ladder.
#
# Nothing here re-reads num_divisions or wif_bat_used mid-loop. Those do not
# refresh inside a single effect, which is what made the old factory
# enforcement pass eat every arms factory in the country. The allowance is
# computed once, up front, and from then on only ever decremented.
wif_disband_over_cap = {
	if = {
		limit = {
			is_ai = yes
			check_variable = { wif_battalions_over > 0 }
			NOT = { has_country_flag = wif_bat_grace }
			check_variable = { num_divisions > 0 }
		}

		# The country, reachable from inside the state loop. The neighbour test
		# below is two scopes deep and an event target reads far better there
		# than a PREV chain.
		save_event_target_as = wif_disband_country

		# Average line battalions per division, so an overshoot measured in
		# battalions can be turned into a number of divisions. Floored at 1 so
		# the division below cannot blow up.
		set_variable = { wif_disband_avg = wif_bat_used }
		divide_variable = { wif_disband_avg = num_divisions }
		if = {
			limit = { check_variable = { wif_disband_avg < 1 } }
			set_variable = { wif_disband_avg = 1 }
		}

		set_variable = { wif_disband_allowance = wif_battalions_over }
		divide_variable = { wif_disband_allowance = wif_disband_avg }
		multiply_variable = { wif_disband_allowance = 0.75 }

		# Pass 1 - the rear. Almost always the only pass that runs, and the only
		# one that can run at all in peacetime.
		every_controlled_state = {
			limit = {
				NOT = {
					any_neighbor_state = {
						controller = { has_war_with = event_target:wif_disband_country }
					}
				}
			}
			wif_disband_state_if_it_fits = yes
		}

		# Pass 2 - the front, and only for a country more than 20% over, the same
		# threshold wif_bat_far_over_cap uses. Reached only when the rear is
		# already dry and allowance is still left, which means the country is deep
		# enough over that the attrition and supply bill costs it more than the
		# divisions are worth.
		#
		# This can only ever fire during a war: at peace every state reads as rear
		# and pass 1 will have spent the allowance first. Deleting this block
		# leaves a rear-only system that still converges once a war ends and every
		# state becomes rear again.
		if = {
			limit = { check_variable = { wif_bat_headroom_pct < -0.2 } }
			every_controlled_state = {
				wif_disband_state_if_it_fits = yes
			}
		}
	}
}


# Clears one state's garrison, but only if that whole garrison fits inside what
# is left of wif_disband_allowance. Called in STATE scope from the loops above,
# so THIS is the state and PREV is the country.
#
# Branch N matches a state holding EXACTLY N divisions - more than N-1, and not
# more than N - which is what makes the subtraction exact. Both bounds are
# needed: without the lower one a small garrison would fall through into a
# larger branch and over-subtract the allowance.
#
# check_variable has no <= or >= operator, only > < =, so "the allowance covers
# N" has to be written as NOT = { allowance < N }. Do not "simplify" it.
wif_disband_state_if_it_fits = {
	if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 0 }
				NOT = { divisions_in_state = { state = PREV size > 1 } }
				NOT = { check_variable = { wif_disband_allowance < 1 } }
			}
		}
		PREV = {
			subtract_from_variable = { wif_disband_allowance = 1 }
			delete_unit = {
				state = PREV
				disband = yes
			}
		}
	}
	else_if = {
		limit = {
			PREV = {
				divisions_in_state = { state = PREV size > 1 }
				NOT = { divisions_in_state = { state = PREV size > 2 } }
