# The German civil war and its aftermath. Effects in
# common/scripted_effects/zz_wif_ger_reunify.txt and zz_wif_ger_abroad.txt.
on_actions = {
	on_weekly = {
		effect = {
			# The last side standing wins the civil war.
			if = {
				limit = { has_country_flag = wif_ger_cw_belligerent }
				wif_cw_check_for_winner = yes
			}
			# A former subject that was at war when the Reich reunited comes
			# back once its war is over.
			if = {
				limit = { has_country_flag = wif_cw_repuppet_after_war }
				wif_cw_repuppet_after_war = yes
			}
		}
	}
	# Free France wins once Vichy is gone.
	on_weekly_FFR = {
		effect = {
			wif_ffr_check_victory = yes
		}
	}
}
