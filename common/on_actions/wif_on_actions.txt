on_actions = {
	# Novgorod's claim dies with Novgorod. Its cores on Novgorod and Luga are
	# stripped whether it capitulates or is annexed outright, so the territory
	# does not stay Novgorod-cored once the republic is finished.
	#
	# Saint Petersburg is deliberately NOT handled here - that core survives and
	# passes to Moskowien in wif_rkm_civilwar.1, because the emergency
	# government was loyalist and its claim is inherited rather than erased.

	# ROOT is the country that capitulated.
	on_capitulation = {
		effect = {
			# Archangelsk falls to the SME.
			#
			# The northern war is a private one - NAR forms with no faction and
			# no guarantees precisely so nobody else can be dragged into it - so
			# capitulation settles it outright rather than dragging on until a
			# peace conference. The SME takes the city and the war ends.
			#
			# Moskowien cannot claw it back afterwards: the fold-back in
			# wif_rkm_civilwar.1 is guarded on country_exists = NAR, and by this
			# point there is no NAR to fold back in.
			if = {
				limit = {
					tag = NAR
					country_exists = SME
					SME = { has_war_with = NAR }
				}
				SME = {
					annex_country = {
						target = NAR
						transfer_troops = no
					}
					add_state_core = 214

					# Vologda goes too, but only if Moskowien's civil war is
					# still running when Archangelsk falls.
					#
					# The northern war and the civil war are separate fights.
					# If the SME finishes its one while Moskowien is still
					# fighting its own, it keeps marching south and takes
					# Vologda off a Reichskommissariat in no state to hold it.
					# Beat the rebels first and the SME stops at Archangelsk.
					#
					# wif_moskowien_civil_war sits on RKM, set in
					# wif_nsr_revolt. Note it is deliberately left set on one
					# of the losing paths (see the comment further down this
					# file), which is the correct reading here too: as long as
					# that war has not been closed out, Moskowien is still
					# distracted.
					#
					# Guarded on RKM still owning it. If a rebel has already
					# taken Vologda then it is not Moskowien's to lose, and
					# prying it out of a third party is a different war.
					if = {
						limit = {
							country_exists = RKM
							RKM = { has_country_flag = wif_moskowien_civil_war }
							351 = { is_owned_by = RKM }
						}
						transfer_state = 351
						add_state_core = 351
					}
				}
			}

			# Moskowien lost the civil war. wif_rkm_civilwar.1 only fires on the
			# winning path, so without this the emergency surrender limit granted
			# in wif_nsr_revolt would never come off.
			#
			# The wif_moskowien_civil_war flag is deliberately left set: if
			# Moskowien is liberated and the rebels are then destroyed, the weekly
			# check should still be able to close the war out properly.
			if = {
				limit = {
					tag = RKM
					has_idea = wif_rkm_civil_war_emergency
				}
				remove_ideas = wif_rkm_civil_war_emergency
			}

			if = {
				limit = { tag = NSR }
				263 = { remove_core_of = NSR }
				208 = { remove_core_of = NSR }
			}
			if = {
				limit = { tag = RLF }
				257 = { remove_core_of = RLF }
				239 = { remove_core_of = RLF }
				255 = { remove_core_of = RLF }
				250 = { remove_core_of = RLF }
			}
			if = {
				limit = { tag = WRS }
				220 = { remove_core_of = WRS }
				240 = { remove_core_of = WRS }
				260 = { remove_core_of = WRS }
				258 = { remove_core_of = WRS }
			}
			if = {
				limit = { tag = SRG }
				241 = { remove_core_of = SRG }
				243 = { remove_core_of = SRG }
			}
			if = {
				limit = { tag = NRG }
				209 = { remove_core_of = NRG }
				210 = { remove_core_of = NRG }
			}
		}
	}

	# ROOT is the winner, FROM is the country being annexed. Covers Moskowien
	# integrating them, and any other victor taking them.
	on_annex = {
		effect = {
			if = {
				limit = { FROM = { tag = NSR } }
				263 = { remove_core_of = NSR }
				208 = { remove_core_of = NSR }
			}
			if = {
				limit = { FROM = { tag = RLF } }
				257 = { remove_core_of = RLF }
				239 = { remove_core_of = RLF }
				255 = { remove_core_of = RLF }
				250 = { remove_core_of = RLF }
			}
			if = {
				limit = { FROM = { tag = WRS } }
				220 = { remove_core_of = WRS }
				240 = { remove_core_of = WRS }
				260 = { remove_core_of = WRS }
				258 = { remove_core_of = WRS }
			}
			if = {
				limit = { FROM = { tag = SRG } }
				241 = { remove_core_of = SRG }
				243 = { remove_core_of = SRG }
			}
			if = {
				limit = { FROM = { tag = NRG } }
				209 = { remove_core_of = NRG }
				210 = { remove_core_of = NRG }
			}
		}
	}

	# Record every country's starting military factories so the ratio cap can
	# grandfather them. A nation that begins the game over its ratio keeps what
	# it started with; only factories built afterwards can ever be taken away.
	on_startup = {
		effect = {
			every_country = {
				# Sub-ideology statistics, so 1936 is right from the first day
				# rather than only at the first weekly tick.
				wif_apply_subideology_modifiers = yes

				# City province counts, seeded onto the states themselves.
				# Generated by tools/city_provinces.py into
				# common/scripted_effects/wif_city_provinces.txt, because HoI4
				# has no province scope and a state cannot be asked what its
				# provinces are made of.
				#
				# THIS HAS TO SIT INSIDE every_country. It was outside it, at
				# the top of the on_startup effect, where the scope is None -
				# and the game refused it with "wif_seed_city_provinces:
				# Invalid Scope ... provided: None", so no state ever got a
				# count and every cored state was worth the no-city rate.
				#
				# The global flag keeps it to one pass rather than one per
				# country. Province membership cannot change during a game, so
				# once is enough.
				if = {
					limit = { NOT = { has_global_flag = wif_city_provs_seeded } }
					set_global_flag = wif_city_provs_seeded
					wif_seed_city_provinces = yes
				}

				set_variable = { wif_mil_grandfather = num_of_military_factories }
				set_country_flag = wif_factory_cap_init
				wif_update_factory_cap = yes

				# The Millions of China. Both exist at game start, so unlike
				# the warlord spirit this one really does fire here.
				if = {
					limit = {
						OR = {
							tag = CHI
							tag = PRC
						}
						NOT = { has_idea = wif_chinese_millions }
					}
					add_ideas = wif_chinese_millions
				}

				# Hungary begins as a regency, because that is what it was.
				#
				# Vanilla starts Horthy on oligarchism with elections allowed.
				# Regential Monarchism is the closer fit - a throne held in trust
				# with no king on it - and it carries no elections, which the
				# effect switches off along with setting the ideology.
				#
				# Done here rather than by shipping a copy of
				# history/countries/HUN - Hungary.txt, which would mean inheriting
				# every future vanilla change to Hungary to alter one line.
				if = {
					limit = {
						tag = HUN
						has_government = neutrality
					}

					wif_become_regential_monarchy = yes
				}

				# Russian Warlord, for completeness. None of the successor
				# states exist at game start, so this normally does nothing -
				# but a scenario that begins with one already on the map would
				# otherwise wait a week for its cap.
				if = {
					limit = {
						wif_is_russian_warlord = yes
						NOT = { has_idea = wif_russian_warlord }
					}
					add_ideas = wif_russian_warlord
				}

				if = {
					limit = {
						wif_is_russian_warlord = yes
						NOT = { has_idea = wif_russian_warlord_economy }
					}
					add_ideas = wif_russian_warlord_economy
				}

				# Closed Economy, forced.
				#
				# Keyed on the SPIRIT rather than on wif_is_russian_warlord, so anything
				# holding it is covered however it got there - a successor state, an
				# event, another mod - not just the tags on the warlord list.
				#
				# closed_economy is a TRADE law, not an economy law, so it sits alongside
				# wif_russian_warlord_economy rather than replacing it. add_ideas swaps
				# out whatever trade law is currently held.
				#
				# Vanilla gates closed_economy behind war, a fascist or communist
				# government and War Economy, none of which a warlord need meet -
				# add_ideas ignores available, and vanilla marks the law
				# cancel_if_invalid = no, so it stays put once granted rather than
				# falling off the moment those conditions are not met.
				#
				# The weekly pass is what makes this a lock: a player who legislates
				# their way out is back on it within seven days, having paid for
				# nothing.
				if = {
					limit = {
						has_idea = wif_russian_warlord
						NOT = { has_idea = closed_economy }
					}
					add_ideas = closed_economy
				}

				# Battalion soft cap. Seeded here so the officer corps readout
				# and the AI both have a cap to read on day one rather than a
				# zero until the first weekly tick.
				wif_recalculate_battalion_cap = yes
			}
		}
	}

	# No enforcement pass. Existing military factories are never removed or
	# converted - the cap is enforced purely by blocking further construction
	# through the wif_military_factory_ceiling spirit.
	#
	# The removed version looped while num_of_military_factories > wif_mil_cap,
	# but that country property does not refresh inside a single effect, so the
	# condition never went false and the loop ran until every arms factory in
	# the country was gone. Do not reintroduce that pattern.

	# Safety net only. Reichskommissariats now receive the law the instant the
	# spirit is granted, via on_add on both Reichskommissariat spirits in
	# common/ideas/GER.txt. This exists for saves where a Reichskommissariat
	# already held the spirit before that hook was added, since on_add cannot
	# fire retroactively. Adding an idea a country already has is a no-op.
	# on_weekly is per-country (ROOT = the country), matching vanilla usage.
	on_weekly = {
		effect = {
			if = {
				limit = {
					OR = {
						has_idea = GER_german_controlled_reichskommissariat
						has_idea = GER_local_controlled_reichskommissariat
					}
					NOT = { has_idea = wif_reichskommissariat_economy }
				}
				add_ideas = wif_reichskommissariat_economy
			}

			# Countries formed after game start never saw on_startup, so
			# grandfather them the first time they are seen here.
			if = {
				limit = { NOT = { has_country_flag = wif_factory_cap_init } }
				set_variable = { wif_mil_grandfather = num_of_military_factories }
				set_country_flag = wif_factory_cap_init
			}

			# No Diplomatic Standing: turn back any volunteers that reached
			# us anyway.
			#
			# The spirit blocks SENDING volunteers with a rule, but nothing
			# in the engine stops a foreign power sending them TO us - there
			# is no receiving-side rule or modifier. recall_volunteers_from
			# is the way round it: it is run by the SENDER against the
			# country they sent to, so this walks every other country and
			# has each one recall from us in turn.
			#
			# The scope shape is copied from vanilla's own use of it in
			# common/decisions/BUL.txt: inside every_other_country, PREV is
			# us, and the tag inside has_volunteers_amount_from is the
			# country being iterated. Swap those two and it silently
			# recalls nothing.
			#
			# Guarantees and incoming lend-lease have no equivalent and are
			# simply not blockable - see the spirit for that note.
			if = {
				limit = { has_idea = wif_splinter_no_factions }
				every_other_country = {
					limit = {
						PREV = {
							has_volunteers_amount_from = {
								tag = PREV
								count > 0
							}
						}
					}
					recall_volunteers_from = PREV
				}
			}

			# The White factions' loyalty, and what it is costing or paying.
			#
			# Recomputed before the civil war check, so the check always reads
			# this week's numbers rather than last week's.
			if = {
				limit = { has_country_flag = wif_watching_the_whites }
				wif_whites_loyalty_update = yes
				wif_whites_civil_war_check = yes
			}

			# The overthrow of the Emergency Government, and what follows it.
			#
			# Ordered after the loyalty recompute above on purpose: the coup
			# reads this week's loyalty to decide whether the handover is
			# bloodless, so it must not read last week's.
			if = {
				limit = { tag = SME }
				wif_sme_coup_check = yes
				wif_russian_federation_check = yes
			}

			# Coring what the Federation takes from Yagoda. Cheap and
			# idempotent, so it simply runs for the rest of the game.
			if = {
				limit = { has_country_flag = wif_is_russian_federation }
				wif_russian_federation_core_check = yes
			}


			# Sub-ideology statistics. Attaches the dynamic modifier that goes
			# with the leader's ideology; each modifier removes itself when the
			# ideology changes, so there is no matching removal here.
			wif_apply_subideology_modifiers = yes

			# The two monarchies that do not hold elections.
			#
			# A sub-ideology cannot carry an election rule - HoI4 has no such key
			# on an ideology or its types, and elections_allowed belongs to
			# set_politics, per country. So the rule is kept here instead of
			# being declared once and trusted.
			#
			# This catches every route in, including a leader whose ideology was
			# changed by another mod's event or by the console, not just the
			# wif_become_* effects in
			# common/scripted_effects/wif_monarchies.txt.
			#
			# Semi-Constitutional Monarchy is deliberately absent: that one keeps
			# its ballot papers, which is the whole difference between it and the
			# other two.
			if = {
				limit = {
					OR = {
						has_country_leader_ideology = wif_regential_monarchism
						has_country_leader_ideology = wif_absolute_monarchism
					}
					has_elections = yes
				}
				set_politics = {
					ruling_party = neutrality
					elections_allowed = no
				}
			}

			# Sweden meets its king.
			#
			# If Sweden has taken SWE_meet_with_the_king and ends up Authoritarian,
			# the arrangement it arrives at is the semi-constitutional one - the
			# crown keeps the substance, the Riksdag keeps the ballot papers.
			#
			# Checked here rather than by editing common/national_focus/sweden.txt,
			# which would mean shipping a copy of the whole Swedish tree and
			# inheriting every future vanilla change to it.
			#
			# The flag makes it a one-off: a Sweden that later changes its mind is
			# not dragged back every week.
			if = {
				limit = {
					# original_tag, not tag: a cosmetic tag or a civil-war
					# successor would slip past a plain tag check.
					original_tag = SWE
					has_completed_focus = SWE_meet_with_the_king
					has_government = neutrality
					NOT = { has_country_flag = wif_swe_semi_constitutional }
				}
				set_country_flag = wif_swe_semi_constitutional
				wif_become_semi_constitutional_monarchy = yes
			}

			# Yagoda rises when the Emergency Government stops being communist
			# enough to be worth obeying.
			#
			# The original route is the war with Archangelsk, seven days after
			# the declaration, from wif_nsr_revolt. This is the second: if the
			# SME's own communist support falls under 49%, the NKVD and the
			# Urals factory managers stop waiting.
			#
			# It fires the EVENT rather than the rising directly, because both
			# of that event's options run wif_yagoda_rising anyway - the rising
			# is automatic, and the choice is only which side the player takes.
			#
			# Guarded twice over: the flag wif_yagoda_risen is set inside the
			# rising itself, and WSM existing means it has already happened.
			if = {
				limit = {
					tag = SME
					communism < 0.49
					NOT = { has_country_flag = wif_yagoda_risen }
					NOT = { country_exists = WSM }
				}
				country_event = { id = wif_sme.4 hours = 6 }
			}

			# The Millions of China, for China and Communist China. Granted the
			# same way and for the same reason as the warlord spirit below: a
			# weekly correction cannot be missed by a spawn route, and this one
			# has to survive a civil war splitting PRC off CHI.
			if = {
				limit = {
					OR = {
						tag = CHI
						tag = PRC
					}
					NOT = { has_idea = wif_chinese_millions }
				}
				add_ideas = wif_chinese_millions
			}

			# Russian Warlord. Held by the successor states of the Soviet
			# collapse - the list is wif_is_russian_warlord in
			# common/scripted_triggers/wif_scripted_triggers.txt - and worth
			# +80% battalion cap through wif_src_warlord.
			#
			# Granted here rather than at each breakaway's creation because
			# there are six of them, spawned from four different scripted
			# effects, and one missed call would be invisible: the country
			# would simply have a smaller army than intended for the rest of
			# the game. A country matching the trigger and lacking the spirit
			# is corrected within a week, whatever route created it.
			#
			# Placed BEFORE the cap recalculation below, so a country that
			# gains the spirit this week gets the cap that goes with it in the
			# same pass rather than a week later.
			#
			# There is no removal branch. Tags do not change, so a country that
			# qualifies once qualifies forever.
			if = {
				limit = {
					wif_is_russian_warlord = yes
					NOT = { has_idea = wif_russian_warlord }
				}
				add_ideas = wif_russian_warlord
			}

			# And the economy that goes with it. Separate from the block above
			# because the two can come apart: a country that already holds the
			# spirit but somehow lost the law - an old save, a scripted law
			# change - is repaired on the next weekly pass. add_ideas on a law
			# swaps whatever economy law is currently held.
			if = {
				limit = {
					wif_is_russian_warlord = yes
					NOT = { has_idea = wif_russian_warlord_economy }
				}
				add_ideas = wif_russian_warlord_economy
			}

			# Closed Economy, forced.
			#
			# Keyed on the SPIRIT rather than on wif_is_russian_warlord, so anything
			# holding it is covered however it got there - a successor state, an
			# event, another mod - not just the tags on the warlord list.
			#
			# closed_economy is a TRADE law, not an economy law, so it sits alongside
			# wif_russian_warlord_economy rather than replacing it. add_ideas swaps
			# out whatever trade law is currently held.
			#
			# Vanilla gates closed_economy behind war, a fascist or communist
			# government and War Economy, none of which a warlord need meet -
			# add_ideas ignores available, and vanilla marks the law
			# cancel_if_invalid = no, so it stays put once granted rather than
			# falling off the moment those conditions are not met.
			#
			# The weekly pass is what makes this a lock: a player who legislates
			# their way out is back on it within seven days, having paid for
			# nothing.
			if = {
				limit = {
					has_idea = wif_russian_warlord
					NOT = { has_idea = closed_economy }
				}
				add_ideas = closed_economy
			}

			# Refresh the cap weekly so the AI strategy reacts promptly when
			# world tension or ideology shifts. Variable maths only, no loops.
			wif_update_factory_cap = yes

			# Battalion soft cap. Also variable maths only in the common case -
			# it walks the country's states only on weeks where the owned or
			# cored state count actually changed. This is the call that applies
			# and removes the overcap penalties.
			wif_recalculate_battalion_cap = yes

			# Then let the AI act on the result. This is the only thing in the
			# mod that removes units, so read the header on
			# wif_disband_over_cap before touching it. AI only; disbanded with a
			# full equipment and manpower refund; rear states before frontline
			# ones; and hard bounded to the overshoot itself, so a country sheds
			# the divisions it is over by and not one more. It no-ops entirely
			# for the player, and for anyone under their cap or still inside the
			# demobilisation grace month.
			wif_disband_over_cap = yes

			# Moskowien has put the civil war down: every breakaway is gone.
			# Announce it, and the news event itself carries out the
			# reunification with the Saint Petersburg emergency government.
			#
			# Add any future civil war combatant to the NOT list below.
			# The war is over when every REBEL is gone. Deliberately not gated on
			# a loyalist still existing: Saint Petersburg or Lokot could be
			# overrun on the way and Moskowien still win.
			if = {
				limit = {
					tag = RKM
					has_country_flag = wif_moskowien_civil_war
					NOT = { country_exists = NSR }
					NOT = { country_exists = SSM }
					NOT = { country_exists = RLF }
					NOT = { country_exists = WRS }
					NOT = { country_exists = SRG }
					NOT = { country_exists = NRG }
				}
				news_event = { id = wif_rkm_civilwar.1 }
			}
		}
	}

	on_state_control_changed = {
		effect = {
			# ROOT is the new controller, so this reads on whoever just took a
			# state. The effect works out for itself whether that mattered.
			wif_ryazan_liberation_check = yes

			219 = {
				if = {
					limit = {
						RKM = { has_country_flag = RKM_moskau_renamed }
						NOT = {
							is_controlled_by = GER
						}
						NOT = {
							is_controlled_by = RKM
						}
					}
					RKM = {
						country_event = { id = wif_rkm.1 }
					}
				}
			}
		}
	}

	# A war between two Russian successor states is not a world event.
	#
	# The Union tearing itself apart should not drag the rest of the planet
	# towards war: Moscow fighting Irkutsk is a Russian matter, and the tension
	# it generates would otherwise unlock wargoals and faction moves for every
	# country in the game that has nothing to do with it.
	#
	# ROOT is the attacker and FROM the defender in on_declare_war - the same
	# scopes vanilla uses in 00_on_actions.txt.
	#
	# HOW MUCH. HoI4 has no modifier that suppresses the tension a war
	# generates, so the tension is cancelled after the fact instead. The engine
	# adds NDefines.NCountry.TENSION_CB_WAR = 10 for a declaration made with a
	# wargoal, which is what every one of these scripted declarations is, and
	# TENSION_NO_CB_WAR = 14 without one. The 10 here matches the first case.
	#
	# It is deliberately a NAMED threat: named sources show up in the world
	# tension breakdown, so the discount is visible to the player rather than
	# being an unexplained dip.
	#
	# TWO THINGS TO WATCH IN GAME, because neither can be settled by reading:
	#   * TENSION_TIME_SCALE_MONTHLY_FACTOR scales the engine's own tension
	#     values down over time, to a floor of 0.25. If it does not scale this
	#     effect the same way, a late-war declaration will over-refund and
	#     world tension will visibly dip. Lower the 10 if that shows up.
	#   * Only the declaration is refunded. If tension turns out to accrue
	#     while these wars are fought rather than only when they start, this
	#     needs a periodic pass rather than a one-off.
	on_declare_war = {
		effect = {
			if = {
				limit = {
					wif_is_russian_warlord = yes
					FROM = { wif_is_russian_warlord = yes }
				}
				add_named_threat = {
					threat = -10
					name = wif_threat_russian_civil_war
				}
			}
		}
	}

	# The Moskowien terror raid countdown.
	#
	# wif_terror_days is set wherever the next raid used to be scheduled - the
	# focus that starts the cycle, and every branch of wif_rkm.20 that schedules
	# the following one. This counts it down and fires the raid at zero.
	#
	# WHY IT IS A COUNTDOWN AND NOT A SCHEDULED EVENT. country_event queues an
	# event that nothing can cancel or bring forward, so the SME's Arm the
	# Partisans decision could never have shortened it. A variable can be halved.
	#
	# Per-tag, so this runs for one country rather than every country in the
	# game. check_variable has no <= operator, hence NOT = { > 0 } for "has
	# reached zero".
	on_daily_RKM = {
		effect = {
			if = {
				limit = { check_variable = { wif_terror_days > 0 } }
				subtract_from_variable = { wif_terror_days = 1 }

				if = {
					limit = { NOT = { check_variable = { wif_terror_days > 0 } } }
					country_event = { id = wif_rkm.20 hours = 6 }
				}
			}
		}
	}

	# Sweden, the moment it turns Authoritarian.
	#
	# The weekly pass alone was not enough: on_weekly fires every seven days, so
	# a Sweden that met its king and changed government on day 6 saw nothing
	# happen at all. This catches the party change itself.
	#
	# Both routes are needed and neither is redundant:
	#   here          the focus is already done and the government changes
	#   on_weekly     the government is already Authoritarian and the focus is
	#                 taken afterwards
	# The shared flag means whichever arrives first wins and the other does
	# nothing.
	on_ruling_party_change = {
		effect = {
			wif_apply_subideology_modifiers = yes

			if = {
				limit = {
					original_tag = SWE
					has_completed_focus = SWE_meet_with_the_king
					has_government = neutrality
					NOT = { has_country_flag = wif_swe_semi_constitutional }
				}
				set_country_flag = wif_swe_semi_constitutional
				wif_become_semi_constitutional_monarchy = yes
			}
		}
	}
}
