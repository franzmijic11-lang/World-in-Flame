# Speer's Economic Council - the quota decisions opened by Quotas for the
# Reichskommissariate (common/national_focus/wif_ger_victory.txt), and closed
# for good by The East Is Bled Dry.
#
# One decision per eastern RK, while it is a German subject. 50 political
# power, or 75 once the SS Economic Empire has taken its share of the labour;
# the cost is paid in complete_effect so it can depend on that.
wif_ger_economic_council_category = {

	wif_ger_requisition_rko = {
		icon = generic_industry

		visible = {
			has_completed_focus = GER_quotas_for_the_reichskommissariate
			NOT = { has_completed_focus = GER_the_east_is_bled_dry }
			country_exists = RKO
			RKO = { is_subject_of = GER }
		}

		available = {
			RKO = { is_subject_of = GER }
		}

		custom_cost_trigger = {
			if = {
				limit = { has_idea = wif_ger_ss_enterprises }
				has_political_power > 74.99
			}
			else = {
				has_political_power > 49.99
			}
		}
		custom_cost_text = wif_ger_quota_cost

		days_re_enable = 365

		complete_effect = {
			if = {
				limit = { has_idea = wif_ger_ss_enterprises }
				add_political_power = -75
			}
			else = {
				add_political_power = -50
			}
			custom_effect_tooltip = wif_ger_cg_minus_1_tt
			hidden_effect = {
				set_variable = { wif_ger_cg_cut = 0.01 }
				wif_ger_facade_cut = yes
			}
			RKO = {
				add_timed_idea = { idea = wif_rk_requisitioned days = 365 }
				add_manpower = -25000
			}
		}

		ai_will_do = {
			factor = 1
