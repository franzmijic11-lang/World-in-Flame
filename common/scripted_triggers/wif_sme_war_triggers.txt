# Country scope: holds either Russian Warlord spirit - the ordinary one, or the
# harsher one the SME and its breakaways get when the SME collapses. Everything
# keyed on being a warlord (the economy law, Closed Economy, the battalion cap)
# reads this rather than the idea, so both count.
wif_has_warlord_spirit = {
	OR = {
		has_idea = wif_russian_warlord
		has_idea = wif_russian_warlord_broken
	}
}

	}
}
