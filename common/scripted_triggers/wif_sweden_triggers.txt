# Who is allowed to hold office after the 1936 election.
#
# The election does not merely decide who governs, it decides who is
# EMPLOYABLE. A Social Democratic ministry does not hire Hogern's front bench,
# and a bourgeois one does not hire Wigforss - so the losing side's politicians
# are locked out of the advisor slots for the rest of the game.
#
# Both read the flags set by the three result focuses in
# common/national_focus/sweden.txt:
#
#   SWE_1936_socialdemokraterna_won
#   SWE_1936_bondeforbundet_won
#   SWE_1936_hogern_won
#
# WRITTEN AS "ALLOWED", NOT "LOCKED", so they drop straight into an advisor's
# available block as one more condition. Before the election none of the flags
# is set, both triggers pass, and every advisor behaves exactly as vanilla
# intends - which matters, because these are added to vanilla characters in
# common/characters/SWE.txt and must be inert until the election resolves.
#
# ACADEMICS ARE NOT POLITICIANS. Alva and Gunnar Myrdal were Social Democrats
# and are deliberately NOT locked: they are hired in this game as social
# scientists rather than as party men, they are gated behind
# SWE_education_reform already, and a bourgeois government had every reason to
# keep them on. Add them here if that judgement should change.

# Passes unless a bourgeois party won. Put on Social Democratic politicians.
wif_swe_socialist_ministry_allowed = {
	NOT = {
		OR = {
			has_country_flag = SWE_1936_bondeforbundet_won
			has_country_flag = SWE_1936_hogern_won
		}
	}
}

# Passes unless the Social Democrats won. Put on Hogern, Bondeforbundet and
# Folkpartiet politicians.
wif_swe_bourgeois_ministry_allowed = {
	NOT = {
		has_country_flag = SWE_1936_socialdemokraterna_won
	}
}

# Has the government kept its promises?
#
# The Democratic War Cabinet is not formed until FOUR of the coalition
# concessions are done, and because each branch holds exactly four, that is the
# same as saying the government finished its own programme.
#
# WRITTEN AS THREE COMPLETE BRANCHES rather than as a count, because HoI4 has no
# way to count completed focuses in a trigger - there is no focus iterator and a
# trigger cannot do arithmetic into a variable. Spelling out the three sets is
# exact, reads clearly in the tooltip, and cannot drift out of step with the
# tree the way a hand-maintained tally would.
#
# The two shared promises appear in both bourgeois branches, which is correct:
# Folkpartiet and Hogern sit in either ministry and collect their price from
# whichever is in office.
#
# ADD ANY NEW CONCESSION FOCUS TO ITS BRANCH HERE, or the branch it belongs to
# will still open the war cabinet one promise short.
wif_swe_four_promises_kept = {
	OR = {
		# Socialdemokraterna
		AND = {
			has_completed_focus = SWE_dyrortsgraderade_folkpensioner
			has_completed_focus = SWE_fri_modravard
			has_completed_focus = SWE_sami_reconcilliation
			has_completed_focus = SWE_form_rfsu
		}

		# Bondeforbundet - two of its own, two shared
		AND = {
			has_completed_focus = SWE_lika_folkpension
			has_completed_focus = SWE_tryggade_jordbrukspriser
			has_completed_focus = SWE_nykterhet_och_folkbildning
			has_completed_focus = SWE_stod_at_forsvarsrorelsen
		}

		# Hogern - two of its own, the same two shared
		AND = {
			has_completed_focus = SWE_sparsamhet_sunda_statsfinanser
			has_completed_focus = SWE_ett_starkare_forsvarsbeslut
			has_completed_focus = SWE_nykterhet_och_folkbildning
			has_completed_focus = SWE_stod_at_forsvarsrorelsen
		}
	}
}
