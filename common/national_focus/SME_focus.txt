# Focus tree for the Soviet Military Emergency Government.
#
# The SME is what the Red Army leaves behind when the Union falls: a general
# staff holding a country it never asked for. Its tree opens on the one piece
# of unfinished business the coup left - Stalin was not in Moscow when the
# army came for him.
#
# The trunk sits at x = 10 rather than x = 0, and every focus on it carries the
# same offset so the column stays vertical - move one, move them all. The gap to
# the left is deliberate, so branches can be hung there later without
# renumbering everything. Two focuses are already off the trunk: the rail pair
# below Industries of Siberia, positioned relatively, and Address the Fracturing
# Leadership at an absolute x = 14.
#
# COST IS IN WEEKS, not days. FOCUS_POINT_DAYS = 7 in vanilla defines and
# Welt in Flammen does not override it, so cost = 9 is exactly 63 days. RKM's
# tree uses the same convention and carries fractional costs (5.71, 8.57, 14.3)
# where a round day count was wanted.

focus_tree = {
	id = SME_focus_tree

	country = {
		factor = 0

		modifier = {
			add = 10
			tag = SME
		}
	}

	default = no

	# =====================================================================
	# Hunt for Stalin and his loyalists
	# =====================================================================
	# The root of the tree. The manhunt always fails - completing it fires
	# wif_sme.2, which reveals that Stalin reached Irkutsk ahead of the army
	# and creates the Far Eastern Union around him.
	#
	# The failure is the point. The SME does not get to choose whether it has
	# a rival; it only gets to find out how far away he is.
	focus = {
		id = SME_hunt_for_stalin
		icon = GFX_focus_SOV_behead_the_snake

		x = 10
		y = 0

		cost = 9			# 63 days

		# No `available` gate. The tree is already SME-only through the
		# country = { } block above, and the SME does not exist until the coup
		# has happened, so there is nothing left to gate on.

		completion_reward = {
			country_event = { id = wif_sme.2 hours = 6 }
		}
	}

	# =====================================================================
	# Deploy East under Marshal Popov
	# =====================================================================
	# THIS FOCUS LIES TO THE PLAYER, on purpose.
	#
	# The description promises a march east to put down Stalin's uprising, and
	# effect_tooltip shows a declaration of war on the FES so the focus panel
	# displays exactly that. Neither happens. The real effect is hidden: Popov
	# takes the army group he was handed for the campaign, keeps going past
	# Irkutsk, and declares the Central Siberian Republic in Tomsk.
	#
	# effect_tooltip renders its contents WITHOUT executing them, which is the
	# whole mechanism here - it is the standard way vanilla shows a consequence
	# it intends to fake or to resolve elsewhere. The real work sits in
	# hidden_effect so no second tooltip gives the game away.
	focus = {
		id = SME_deploy_east_under_popov
		icon = GFX_goal_generic_major_war

		prerequisite = { focus = SME_hunt_for_stalin }

		x = 10
		y = 1

		cost = 7.14			# 50 days

		completion_reward = {
			# The promise. Shown, never run.
			effect_tooltip = {
				declare_war_on = {
					target = FES
					type = annex_everything
				}
			}

			# What actually happens.
			hidden_effect = {
				country_event = { id = wif_sme.3 hours = 6 }
			}
		}
	}

	# =====================================================================
	# Prepare to Take Back What the Germans Took
	# =====================================================================
	# The Emergency Government turns the industry it still holds towards a war
	# it has not declared. Nothing here is military: the west is not coming back
	# until there is something to take it back with.
	#
	# The three bonuses live on the national spirit
	# wif_sme_prepare_to_take_back in common/ideas/wif_national_spirits.txt.
	# A focus runs effects and cannot hold a country modifier itself, so the
	# completion reward is the spirit and nothing else.
	focus = {
		id = SME_prepare_to_take_back
		icon = GFX_focus_generic_military_industry

		prerequisite = { focus = SME_deploy_east_under_popov }

		x = 10
		y = 2

		cost = 10			# 70 days

		search_filters = { FOCUS_FILTER_INDUSTRY }

		completion_reward = {
			add_ideas = wif_sme_prepare_to_take_back
		}
	}

	# =====================================================================
	# Address the Fracturing Leadership
	# =====================================================================
	# The political answer to Popov, and the only focus on this tree that hangs
	# off the trunk instead of continuing it. It branches from Deploy East - the
	# focus that caused the problem - and sits at the same depth as Prepare to
	# Take Back, so the two are alternatives for the same stretch of the game
	# rather than one gating the other.
	#
	# x = 14, not the trunk's 10. Four columns clear - well past the
	# right-hand rail branch at x = 11, with room between the two for the
	# political side of the tree to grow into.
	#
	# It does not repair the fractured government, it only makes it liveable:
	# the spirit granted by wif_sme.3 is swapped for a milder copy of itself.
	# No has_idea guard on the swap - the prerequisite IS the focus that fires
	# wif_sme.3, the event lands six hours after that focus completes, and this
	# one takes fifty days, so the spirit is always on the country by the time
	# the swap can run.
	focus = {
		id = SME_address_fracturing_leadership
		icon = GFX_goal_generic_political_pressure

		prerequisite = { focus = SME_deploy_east_under_popov }

		x = 14
		y = 2

		cost = 7.14			# 50 days

		search_filters = { FOCUS_FILTER_POLITICAL }

		completion_reward = {
			swap_ideas = {
				remove_idea = wif_sme_fractured_government
				add_idea = wif_sme_fractured_government_addressed
			}

			# The question the focus is really about: who the Emergency
			# Government is. set_default = yes makes this the balance the
			# power-balance button opens on.
			#
			# NO set_value HERE. The game logs "both set_default and set_value
			# are specified, will use set_value!" and then ignores set_default,
			# which would leave the balance created but not the country's
			# default one. The starting position is initial_value = 0 in
			# common/bop/SME.txt instead, which is dead centre either way.
			set_power_balance = {
				id = SME_leadership_balance
				left_side = SME_leadership_council_side
				right_side = SME_leadership_zhukov_side
				set_default = yes
			}
		}
	}

	# =====================================================================
	# Industries of Siberia
	# =====================================================================
	# The build-up the previous focus only promised. Four states east of the
	# armistice line get the plant, and the room to keep expanding it - the slots
	# matter as much as the factories, because a state at its building limit
	# cannot grow no matter how much construction the country has.
	#
	# STATE IDS resolved through the STATE_<id> localisation, never filenames:
	#
	#   403 Tyumen      653 Sverdlovsk      249 Kazan      397 Syktyvkar
	#
	# instant_build = yes throughout: this is a reward for finishing the focus,
	# not a construction queue for the player to pay for a second time.
	#
	# Nothing here is guarded on ownership. Every one of the four is SME core
	# territory - 249, 397 and 403 are in the heartland cored by wif_sov.1, and
	# Sverdlovsk is the Urals industry the WSM breaks away with - so if the SME
	# has lost them by this point the effect quietly does nothing to states it no
	# longer holds, which is the correct outcome anyway.
	focus = {
		id = SME_industries_of_siberia
		icon = GFX_focus_generic_industry_3

		prerequisite = { focus = SME_prepare_to_take_back }

		x = 10
		y = 3

		search_filters = { FOCUS_FILTER_INDUSTRY }

		completion_reward = {

			# being built as an arsenal.

				add_extra_state_shared_building_slots = 4
				add_building_construction = {
					type = arms_factory
					level = 4
					instant_build = yes
				}
			}

			# Sverdlovsk - the balanced one, arms and civilian both.

				add_extra_state_shared_building_slots = 4
				add_building_construction = {

				}

					type = industrial_complex

			249 = {
				add_extra_state_shared_building_slots = 2
				add_building_construction = {
					type = arms_factory
					level = 1
					instant_build = yes
				}
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}

			397 = {
				add_extra_state_shared_building_slots = 2
				add_building_construction = {
					type = arms_factory
					level = 1
					instant_build = yes
				}
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
		}
	}

	# =====================================================================
	# The two rail branches off Industries of Siberia
	# =====================================================================
	# Both are positioned RELATIVELY off SME_industries_of_siberia rather than
	# with the absolute x = 10 the trunk uses, so moving the trunk carries them.
	#
	# ROUTES ARE PINNED, not left to the pathfinder.
	#
	# The first version handed build_railway a start and target province and let
	# it choose. It never got the chance: every leg was wrapped in
	# NOT = { has_railway_connection }, and the focus itself bypassed on the same
	# test. Kuybyshev, Kazan and Ufa ARE already rail-connected in 1936 - by a
	# long way round through the pre-war network - so the trigger answered "yes,
	# connected", the bypass fired and not a metre of track was laid.
	#
	# has_railway_connection cannot express "connected DIRECTLY". It answers
	# only whether a route exists at any length, and has_railway_level is
	# state-wide, so neither can carry the guard. Both are therefore gone: these
	# focuses now simply build their lines. A stretch that already has rail on it
	# gains a level - that is the cost of having the line appear at all.
	#
	# The province lists below are shortest paths computed off provinces.bmp
	# (4-neighbour adjacency, A* biased by centroid distance so the line stays
	# straight). Each was checked to stay inside the states it connects:
	#
	#   Kazan - Ufa            11 provinces, Kazan > Ufa
	#   Kazan - Kuybyshev       7 provinces, Kazan > Kuybyshev
	#   Magnitogorsk - Ufa      5 provinces, Magnitogorsk > Ufa
	#   Kazan - Kirov           7 provinces, Kazan > Mari El > Kirov
	#   Kirov - Syktyvkar       6 provinces, Kirov > Syktyvkar
	#   Syktyvkar - 3147        5 provinces, Syktyvkar > Pechora > Nenets
	#
	# That last one is why the northern leg needs no separate Pechora step: the
	# straight line from Syktyvkar to the Nenets target runs through Pechora on
	# its own.
	#
	# EVERY LEG CARRIES fallback = yes AND ITS ENDPOINTS. The Kazan - Kirov line
	# silently built nothing on the first attempt: a pinned path is all or
	# nothing, and if the engine disagrees with one step of it the whole call is
	# discarded without a word in the log. Its first hop, 6338 to 11387, shares
	# about seven pixels of border - the sort of near-corner contact that need
	# not be adjacency in the game's own graph. Border width turned out not to
	# separate the working legs from the broken one (Kazan - Ufa gets by on
	# eight), so rather than keep guessing at the engine's rule, fallback = yes
	# makes the question moot: the path is a preference, and if it is refused
	# the line is still built between start_province and target_province.

	# --- right branch ----------------------------------------------------
	focus = {
		id = SME_connect_south
		icon = GFX_goal_generic_construct_infrastructure

		prerequisite = { focus = SME_industries_of_siberia }

		relative_position_id = SME_industries_of_siberia
		x = 1
		y = 1

		cost = 7.14			# 50 days

		search_filters = { FOCUS_FILTER_INDUSTRY }

		completion_reward = {
			# Kazan - Ufa
			build_railway = {
				level = 1
				fallback = yes
				path = { 6338 372 9514 11496 355 3328 11389 6308 9349 273 1278 }
				start_province = 6338
				target_province = 1278
			}
			# Kazan - Kuybyshev
			build_railway = {
				level = 1
				fallback = yes
				path = { 6338 372 9514 292 6404 339 11225 }
				start_province = 6338
				target_province = 11225
			}
			# Magnitogorsk - Ufa
			build_railway = {
				level = 1
				fallback = yes
				path = { 10256 4311 4265 1258 1278 }
				start_province = 10256
				target_province = 1278
			}
			# The Magnitogorsk works get the ore and the fuel to go with the line.
			add_resource = {
				type = oil
				amount = 2
				state = 582
			}
			add_resource = {
				type = steel
				amount = 6
				state = 582
			}
		}
	}

	focus = {

		icon = GFX_goal_generic_construct_infrastructure

		prerequisite = { focus = SME_industries_of_siberia }

		relative_position_id = SME_industries_of_siberia

	# =====================================================================
	# Zlatoust Ural Military District
	# =====================================================================
	# Where the two rail branches meet. It needs BOTH - two separate
	# prerequisite blocks, which HoI4 reads as AND, not as a choice. One
	# prerequisite block listing both focuses would mean "either will do".
	#
	# Positioned at x = 0 off Industries of Siberia, one row below the rail
	# pair that sits at x = -1 and x = +1, so it closes the diamond they open.
	#
	# STATE 573 is Zlatoust, resolved through STATE_573 in the localisation
	# rather than off a filename.
	#
	# The slots come first in the list for a reason: five arms factories will
	# not fit in a state that has no room for them, and add_building_construction
	# does not create space of its own.
	#
	# Not guarded on ownership, the same as Industries of Siberia above: if the

	# =====================================================================

	# =====================================================================
	# Who the NKVD answers to
	# =====================================================================
	# Mutually exclusive, and the second argument the balance of power hosts.
	# Both need SME_address_fracturing_leadership as well as the NKVD focus, but
	# only the NKVD focus is a prerequisite - see the note on available below.
	#
	# Ten points each, against the eleven of the pair above - so a player who
	# takes one focus from each pair on the same side clears the neutral band
	# comfortably, and one from each side very nearly cancels out.
	#
	# 15 days, deliberately short. The argument about who the Commissariat
	# answers to is settled the moment it is asked out loud; the work was all in
	# rebuilding the thing in the first place.
	#
	# The spirit reward is identical on both paths: the NKVD works harder for the
	# Party once it knows whose it is, and it does not much matter which answer
	# it got. What differs is who ends up holding the leash.
	focus = {
		id = SME_marshals_secret_police
		icon = GFX_focus_generic_secret_agent_operative

		prerequisite = { focus = SME_rebuild_the_nkvd }

		# Address the Fracturing Leadership is required, but as an AVAILABILITY
		# condition rather than a prerequisite, so the tree draws no line back to
		# it. A second prerequisite block would have been an AND and would have
		# worked, but it would also have run an arrow across the branch.
		#
		# The requirement is not cosmetic: add_power_balance_value does nothing
		# at all if the balance does not exist, and that focus is what creates
		# it. Without this the focus could be taken early and the reward would
		# be thrown away in silence.
		available = {
			has_completed_focus = SME_address_fracturing_leadership
		}

		mutually_exclusive = { focus = SME_make_them_auditable }

		relative_position_id = SME_rebuild_the_nkvd
		x = -1
		y = 1

		cost = 2.14			# 15 days

		search_filters = { FOCUS_FILTER_POLITICAL }

		completion_reward = {
			swap_ideas = {
				remove_idea = wif_nkvd
				add_idea = wif_nkvd_marshals_police
			}

			add_power_balance_value = {
				id = SME_leadership_balance
				value = 0.10
				tooltip_side = SME_leadership_zhukov_side
			}
		}
	}

	focus = {
		id = SME_make_them_auditable
		icon = GFX_focus_generic_court
