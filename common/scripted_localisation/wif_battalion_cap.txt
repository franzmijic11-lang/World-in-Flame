# Tooltip text for the battalion cap readout in the officer corps screen.
#
# Localisation cannot branch, cannot loop and cannot hide a line, which is why
# every conditional word in that tooltip has to come through here.
#
# TWO KINDS OF ENTRY LIVE IN THIS FILE
#
# 1. The breakdown lines. One defined_text per bonus, each with exactly two
#    branches: the bonus is live, or it is not. A bonus that is not live
#    resolves to wif_cap_line_none, which is a genuinely empty string, so the
#    line disappears from the tooltip entirely rather than showing "+0%".
#    That is the whole point - the player should see the seven things paying
#    for their army, not eighteen rows of which eleven are zero.
#
#    Every one of these reads the same wif_src_* variable the cap maths wrote
#    in common/scripted_effects/wif_battalion_cap.txt. They cannot drift out of
#    agreement with the real number, because there is only one number.
#
# 2. The penalty breakdown, which was here first and is unchanged.
#
# ADDING A NEW BONUS TO THE CAP means four edits, and missing any of them
# leaves the tooltip quietly lying: bank it in a wif_src_* variable, zero that
# variable at the top of the pass, add it to the summation block, then add a
# defined_text here and its line to the localisation file.


# The conscription law's own name, so the tooltip can say WHICH law is paying
# out rather than just how much. Ordered exactly like the cap maths, and the
# final branch catches Disarmed Nation, which contributes nothing.
defined_text = {
	name = GetWifCapLawName

	text = {
		trigger = { has_idea = scraping_the_barrel }
		localization_key = wif_cap_law_barrel
	}

	text = {
		trigger = { has_idea = all_adults_serve }
		localization_key = wif_cap_law_adults
	}

	text = {
		trigger = { has_idea = service_by_requirement }
		localization_key = wif_cap_law_requirement
	}

	text = {
		trigger = { has_idea = extensive_conscription }
		localization_key = wif_cap_law_extensive
	}

	text = {
		trigger = { has_idea = limited_conscription }
		localization_key = wif_cap_law_limited
	}

	text = {
		trigger = { has_idea = volunteer_only }
		localization_key = wif_cap_law_volunteer
	}

	text = {
		trigger = { always = yes }
		localization_key = wif_cap_law_disarmed
	}
}

# Economy law. THE ONLY LINE IN THIS TOOLTIP THAT CAN BE A PENALTY, so unlike
# every other bonus it has three branches rather than two: paying out, costing,
# or absent. The two live branches point at different localisation keys because
# the sign and the colour both have to change - a penalty renders red and
# carries its own minus, and the "+" the other lines hardcode would otherwise
# print as "+-15%".
#
# Absent means Early Mobilisation, the 0% law, which is hidden exactly like
# Disarmed Nation is on the conscription line.
defined_text = {
	name = GetWifCapEconomy

	text = {
		trigger = { check_variable = { wif_src_econ > 0 } }
		localization_key = wif_cap_line_econ
	}

	text = {
		trigger = { check_variable = { wif_src_econ < 0 } }
		localization_key = wif_cap_line_econ_neg
	}

	text = {
		trigger = { always = yes }
		localization_key = wif_cap_line_none
	}
}

# Balance of strength. Three live branches rather than the usual one, because
# the clause pays out in every war - what changes is which band, so the tooltip
# has to name the band rather than merely show a number. Ordered weakest enemy
# first, matching the cap maths.
defined_text = {
	name = GetWifCapWarBalance

	text = {
		trigger = {
			check_variable = { wif_src_war_bal > 0 }
			check_variable = { wif_val_enemy_ratio < 0.9 }
		}
		localization_key = wif_cap_line_war_weaker
	}

	text = {
		trigger = {
			check_variable = { wif_src_war_bal > 0 }
			check_variable = { wif_val_enemy_ratio > 1.1 }
		}
		localization_key = wif_cap_line_war_stronger
	}

	text = {
		trigger = { check_variable = { wif_src_war_bal > 0 } }
		localization_key = wif_cap_line_war_even
	}

	text = {
		trigger = { always = yes }
		localization_key = wif_cap_line_none
	}
}

# Penalty breakdown, unchanged. Three states, checked in order - the first
# matching branch wins, so the grace clause has to come before the plain
# over-cap one:
#
#   1. Over the cap, but the one month demobilisation grace is still running.
#      Nothing is being applied yet, and saying so is the whole point of
#      having a grace period.
#   2. Over the cap with the grace expired. The dynamic modifier is on and
#      these are the live numbers, read straight from the same variables the
#      modifier itself reads.
#   3. Within the cap. No penalties.

defined_text = {
	name = GetWifBattalionCapPenalties

	text = {
		trigger = {
			check_variable = { wif_battalions_over > 0 }
			has_country_flag = wif_bat_grace
		}
		localization_key = wif_cap_penalties_grace
	}

	text = {
		trigger = {
			check_variable = { wif_battalions_over > 0 }
		}
		localization_key = wif_cap_penalties_active
	}

	text = {
		trigger = { always = yes }
		localization_key = wif_cap_penalties_none
	}
}
