# Swedish national spirits.
#
# THE FILENAME IS NOT sweden.txt ON PURPOSE. That name would override vanilla's
# common/ideas/sweden.txt wholesale and delete every Swedish idea in it -
# SWE_gustaf_v_idea included, which is the one the war cabinet swaps FROM. A
# distinct filename merges into the same groups instead, and because every key
# below is new there is no duplicate for the engine to discard. (Contrast
# common/ideas/wif_trade_laws.txt, which redefines eight vanilla keys and does
# produce duplicates.)
ideas = {
	country = {

		# ============================================================
		# Gustaf V after the Democratic War Cabinet
		# ============================================================
		# A COUNTRY IDEA CARRYING A LEADER TRAIT, which is how the base game
		# models a monarch as a national spirit - vanilla's SWE_gustaf_v_idea is
		# built exactly this way, with traits = { SWE_former_autocrat }. The
		# blocks below are copied from it so the king behaves identically in
		# every respect except the trait, and in particular keeps the same
		# allowed_civil_war condition: he goes with whichever Sweden is not
		# communist, and not to a rival non-aligned Sweden.
		wif_swe_gustaf_v_constitutional = {

			allowed = {
				original_tag = SWE
			}

			picture = SWE_king_gustaf

			allowed_civil_war = {
				NOT = {
					has_government = communism
				}
				NOT = {
					any_other_country = {
						original_tag = SWE
						has_government = neutrality
						NOT = {
							tag = ROOT
						}
					}
				}
			}

			removal_cost = -1

			# The whole change. See common/country_leader/wif_traits.txt:
			# +5% stability, +10% trade opinion, and none of the autocrat's
			# political baggage.
			traits = { SWE_constitutional_monarch }
		}

		# ============================================================
		# The coalition's price, under an agrarian government
		# ============================================================
		# Four spirits, granted by the four concession focuses beneath Form an
		# Agrarian Cabinet. Each is a policy one of the three coalition parties
		# extracted in exchange for keeping Pehrsson-Bramstorp in the chair, and
		# each one costs the government something - that is what makes them
		# concessions rather than rewards.
		#
		# allowed = { always = no } keeps every one of them off the country at
		# game start; they are only ever granted by their focus.

		# Bondeforbundet's own, and the one it fought hardest for: a flat
		# old-age pension, the same in Kiruna as in Stockholm.
		wif_swe_lika_folkpension = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_pp_unity_bonus

			modifier = {
				stability_factor = 0.03

				# Recruitable Population FACTOR, not the flat conscription key.
				# conscription = 0.02 would add two whole percentage points to
				# the recruitable share of the population, which is more than
				# the gap between Limited and Extensive Conscription and would
				# dwarf every other number on this branch. This is +2% of what
				# the country already draws, which is what the promise is worth.
				conscription_factor = 0.02

				# The bill. A pension paid at the same rate everywhere is a
				# pension paid at the town rate everywhere.
				consumer_goods_factor = 0.02
			}
		}

		# The continuation of the kohandel: milk price equalisation, grain
		# regulation and a tariff wall to hold them both up.
		wif_swe_tryggade_jordbrukspriser = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_agrarian_society

			modifier = {
				# HoI4 HAS NO BUILD-COST MODIFIER FOR A SINGLE BUILDING TYPE -
				# construction is priced in speed, not in cost, so "10% cheaper
				# to build infrastructure" is expressed as 10% faster. The
				# player sees the same thing: roads and rail go up for less.
				production_speed_infrastructure_factor = 0.10

				war_support_factor = 0.05

				# Protectionism cuts both ways. A country that shuts its
				# agricultural market is a country others are less willing to
				# trade with.
				trade_opinion_factor = -0.05
			}
		}

		# Folkpartiet's price. The party of the temperance movement, the free
		# churches and the study circles wants the state behind all three.
		#
		# SPLIT IN TWO ON PURPOSE. The stability is permanent - a settled moral
		# order does not lapse - while the burst of research and the political
		# cost of enforcing sobriety both run 180 days and then stop. One idea
		# could not express that; a timed idea cannot carry a permanent
		# modifier alongside a temporary one.
		wif_swe_nykterhet_och_folkbildning = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_research_bonus

			modifier = {
				stability_factor = 0.02
			}
		}

		# The timed half. Added with add_timed_idea for 180 days, so it expires
		# on its own and nothing has to remember to remove it.
		wif_swe_nykterhetskampanjen = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_research_bonus

			modifier = {
				research_speed_factor = 0.05

				# political_power_gain, NOT political_power_factor: the promise
				# costs a flat 0.05 a day, not a percentage of the daily gain.
				# Moral politics wins no friends in the labour movement.
				political_power_gain = -0.05
			}
		}

		# Hogern's price - the party that thought the 1936 defence act too
		# timid and put its weight behind the shooting clubs, the Lottor and
		# voluntary training instead.
		wif_swe_frivilliga_forsvarsrorelsen = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = citizen_militia_1

			modifier = {
				# Negative is the good direction: divisions spend less time in
				# training because a good many of the men arrive having already
				# done some.
				training_time_factor = -0.05

				war_support_factor = 0.03

				# Arming the bourgeois volunteer movement is read in the union
				# halls as exactly what it looks like, and the government pays
				# for that every time it wants a minister.
				political_advisor_cost_factor = 0.05
			}
		}

		# ============================================================
		# The Social Democratic programme
		# ============================================================
		# The other side of the same 1936 argument. Bondeforbundet wanted the
		# old-age pension paid at one flat rate everywhere; the Social Democrats
		# wanted it graded by cost of living, which is worth more in a town than
		# on a farm. That disagreement is what brought Hansson's government down
		# in June and what the election was fought over.
		#
		# The two spirits are deliberately NOT opposites in game terms - the
		# player can never hold both, because they hang off different cabinets -
		# so each simply does what its party said it would.
		wif_swe_dyrortsgraderade_folkpensioner = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_pp_unity_bonus

			modifier = {
				stability_factor = 0.05

				# political_power_gain, not political_power_factor: a flat
				# +0.05 a day. A government that delivered on the question it
				# fell over is a government with authority.
				political_power_gain = 0.05

				# Grading the pension upwards in the towns costs more than
				# paying one rate everywhere, and the towns are where most of
				# the pensioners are.
				consumer_goods_factor = 0.03
			}
		}

		# Free maternity care, midwives across the country, and money for new
		# mothers. The Riksdag legislated moderskapspenning and modrahjalp in
		# 1937, immediately after the election.
		#
		# Split in two the same way Folkbildning is: the reform is permanent,
		# the cost of pushing it through the chamber is not.
		wif_swe_fri_modravard = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_pp_unity_bonus

			modifier = {
				monthly_population = 0.03
				stability_factor = 0.02
			}
		}

		# The timed half - the government's own resources and room to negotiate,
		# spent on getting the bill through.
		wif_swe_moderskapsreformen = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_pp_unity_bonus

			modifier = {
				political_power_gain = -0.05
			}
		}

		# Hogern's own, and the one concession an agrarian ministry will not
		# make: a stand against planhushallning, against rising state spending
		# and against the taxes that pay for both. Granted only by a government
		# Hogern actually leads.
		wif_swe_sparsamhet_sunda_statsfinanser = {
			allowed = {
				always = no
			}

			allowed_civil_war = {
				always = yes
			}

			removal_cost = -1

			picture = generic_economic_increase

			modifier = {
				# THE SAME SUBSTITUTION AS THE INFRASTRUCTURE LINE ABOVE, and
				# for the same reason: HoI4 prices construction in SPEED and has
				# no build-cost modifier for any individual building type. A
				# civilian factory 5% cheaper is a civilian factory 5% faster,
				# and the player's money goes exactly as far either way.
				production_speed_industrial_complex_factor = 0.05

				# A tight budget and a defended private sector. Negative is the
				# good direction here - it is the share of civilian industry the
				# country has to spend on itself.
				consumer_goods_factor = -0.02

				# The labour movement reads austerity as an attack on the reform
				# programme, and is not entirely wrong.
				stability_factor = -0.02
			}
		}
	}
}
