# Facade Economy - what MEFO Bills becomes once Autarky Achieved is completed.
#
# A copy of vanilla's GER_mefo_bills_modifier (common/dynamic_modifiers/
# wuw_dynamic_modifiers.txt), line for line, under its own name. It reads the
# SAME GER_industrial_* variables, so every value Mefo Bills had built up -
# and the production bonuses Autarky Achieved adds to them - carries straight
# over. Only the name and the consumer goods behaviour differ.
#
# It has to be a separate modifier and not a rename: vanilla's on_monthly_GER
# grows consumer goods for anything holding GER_mefo_bills_modifier, clamps it
# at 100% and applies the over-100% penalties. Under a name vanilla does not
# know, none of that touches it, and the growth is WiF's own instead - see
# on_monthly_GER in common/on_actions/wif_on_actions.txt.
wif_ger_facade_economy_modifier = {
	enable = { always = yes }
	icon = GFX_idea_ger_mefo_bills
	custom_modifier_tooltip = wif_ger_facade_economy_cg_tt
	consumer_goods_factor = GER_industrial_consumer_goods_factor
	civilian_intel_to_others = GER_industrial_civilian_intel_to_others
	army_intel_to_others = GER_industrial_army_intel_to_others
