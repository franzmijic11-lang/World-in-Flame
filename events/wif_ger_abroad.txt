# America is offered a coup in Britain. Three days in.
country_event = {
	id = wif_ger_abroad.2
	title = wif_ger_abroad.2.t
	desc = wif_ger_abroad.2.d
	picture = GFX_report_event_french_british_officers

	is_triggered_only = yes

	trigger = {
		country_exists = ENG
		ENG = {
			is_subject = yes
			OVERLORD = { wif_cw_is_german_power = yes }
		}
	}

	option = {
		name = wif_ger_abroad.2.a
		ai_chance = {
			base = 90
			# Watch the Generals (tools/gen_eng_protektorat_tree.py): 90% -> 30%
			modifier = {
				factor = 0.0476
				ENG = { has_country_flag = wif_eng_watch_the_generals }
			}
		}
		custom_effect_tooltip = wif_ger_abroad.2.a_tt
		hidden_effect = {
			wif_cw_british_coup = yes
			news_event = { id = wif_ger_abroad.6 }
		}
	}
	option = {
		name = wif_ger_abroad.2.b
		ai_chance = { base = 10 }
	}
}
