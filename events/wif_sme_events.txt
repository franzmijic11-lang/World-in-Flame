# Stalin escaped - fires 6 hours after SME_hunt_for_stalin completes.
#
# The manhunt has no success branch. Completing the focus is how the SME finds
# out where Stalin went, not whether it caught him. The single option carries
# the whole split: twelve far-eastern states leave for the new Far Eastern
# Union, Rokossovsky and Vasilevsky go with them, and Khabarovsk follows
# Vladivostok rather than either government.
country_event = {
	id = wif_sme.2
	title = wif_sme.2.t
	desc = wif_sme.2.desc
	picture = GFX_report_event_generic_train

	is_triggered_only = yes

	# Stay in Moscow.
	option = {
		name = wif_sme.2.a

		ai_chance = {
			base = 100
		}

		wif_stalin_flees_east = yes
	}

	# Go east with him.
	#
	# The split runs FIRST and the tag change second - change_tag_from cannot
	# hand the player a country that does not exist yet, and the FES is not
	# created until wif_stalin_flees_east has transferred Irkutsk. Hence the
	# country_exists guard: if Japan is sitting on the whole Far East then no
	# FES was raised, there is nothing to defect to, and the player simply
	# stays with the SME.
	#
	# base = 0 keeps the AI out of it. This is a player-facing choice only -
	# an AI SME picking it would hand its own country to nobody.
	option = {
		name = wif_sme.2.b

		ai_chance = {
			base = 0
		}

		wif_stalin_flees_east = yes

		hidden_effect = {
			if = {
				limit = { country_exists = FES }
				FES = { change_tag_from = ROOT }
			}
		}
	}
}

# Popov's betrayal - fires 6 hours after SME_deploy_east_under_popov completes.
#
# The player was told this focus would start a war on the FES. Instead the
# marshal sent to fight it has declared his own republic in Tomsk and taken a
# third of the army with him. One option, because there is nothing to decide:
# by the time Moscow hears about it, Central Siberia is already gone.
country_event = {
	id = wif_sme.3
	title = wif_sme.3.t
	desc = wif_sme.3.desc
	picture = GFX_report_event_soviet_purge_officers_

	is_triggered_only = yes

	# Stay in Moscow.
	option = {
		name = wif_sme.3.a

		ai_chance = {
			base = 100
		}

		wif_popov_betrayal = yes
	}

	# Go east with Popov.
	#
	# Same shape as wif_sme.2.b: the split runs first and the tag change
	# second, because change_tag_from cannot hand the player a country that
	# does not exist yet, and the CSR is not created until Tomsk has actually
	# transferred. Hence the country_exists guard - if the SME no longer owns
	# a single one of the ten states then no republic was proclaimed, there is
	# nothing to defect to, and the player stays where they are.
	#
	# base = 0 keeps the AI out of it. An AI SME picking this would hand its
	# own country to nobody.
	option = {
		name = wif_sme.3.b

		ai_chance = {
			base = 0
		}

		wif_popov_betrayal = yes

		hidden_effect = {
			if = {
				limit = { country_exists = CSR }
				CSR = { change_tag_from = ROOT }
			}
		}
	}
}

# Yagoda is alive - fires one week after the SME declares war on Archangelsk.
#
# Scheduled from wif_nsr_revolt, off the declaration itself rather than off any
# outcome of the northern war. The trigger block re-checks the SME still exists
# when it lands, since a week is long enough for the situation to have changed.
country_event = {
	id = wif_sme.4
	title = wif_sme.4.t
	desc = wif_sme.4.desc
	picture = GFX_report_event_soviet_purge_trial

	is_triggered_only = yes

	trigger = {
		tag = SME
	}

	# Stay in Moscow.
	option = {
		name = wif_sme.4.a

		ai_chance = {
			base = 100
		}

		wif_yagoda_rising = yes
	}

	# Go over to Yagoda.
	#
	# The rising runs first and the tag change second, for the same reason as
	# wif_sme.2.b and wif_sme.3.b - the WSM does not exist until a state has
	# transferred to it. The country_exists guard covers the case where the
	# SME had already lost all four Urals states, in which case nothing was
	# declared and there is nowhere to go.
	#
	# base = 0 keeps the AI out of it.
	option = {
		name = wif_sme.4.b

		ai_chance = {
			base = 0
		}

		wif_yagoda_rising = yes

		hidden_effect = {
			if = {
				limit = { country_exists = WSM }
				WSM = { change_tag_from = ROOT }
			}
		}
	}
}

# Return of a Romanov. Fired by the decision wif_invite_vladimir_iii.
#
# The Party did not agree to this and is not being asked. Fifteen points of
# stability is what it costs to put a Romanov back on Russian soil while calling
# yourself a workers' state.
#
# Kuybyshev is demilitarised because the whites insisted on it - a Tsar in a
# garrison town is a Tsar with an army around him, which is precisely what the
# people who agreed to this were promised would not happen. The palace is the
# consolation: one extra building slot in the state, and it stays built whatever
# the politics do afterwards.
country_event = {
	id = wif_sme.5
	title = wif_sme.5.t
	desc = wif_sme.5.d
	picture = GFX_report_event_generic_rally

	is_triggered_only = yes

	option = {
		name = wif_sme.5.a

		add_stability = -0.15

		251 = {
			set_demilitarized_zone = yes

			add_dynamic_modifier = {
				modifier = wif_new_imperial_palace
			}
		}
	}
}

# What Chelyabinsk is going to be for. Fired by SME_chelyabinsk_institute.
#
# Four options, one taken - an event rather than a decision category because
# the choice is a single moment and an option is exclusive by construction.
# Nothing costs anything: the focus was the price.
#
# PROVINCE 9292 IS THE VICTORY POINT of state 572. The state carries fifteen
# special_project_facility_spawn points in map/buildings.txt, and exactly one of
# them sits inside 9292 - the urban province the victory point is on - which is
# why these options can name a province at all. Without a spawn point in the
# named province the construction FAILS SILENTLY, so if 572's province list is
# ever edited, check that spawn is still inside it.
#
# THE FIRST THREE WOULD EXCLUDE EACH OTHER WITHOUT THE EVENT. nuclear_facility,
# air_facility and land_facility all declare
# level_cap = { province_max = 1 state_max = 1 group_by = special_project_facility }
# in common/buildings/00_buildings.txt, so they share a single slot per state.
# The fourth is not a building, and only the single-option rule keeps it from
# stacking on top of one.
country_event = {
	id = wif_sme.6
	title = wif_sme.6.t
	desc = wif_sme.6.desc
	picture = GFX_report_event_generic_factory

	is_triggered_only = yes

	# ----------------------------------------------------------------- nuclear
	option = {
		name = wif_sme.6.a

		572 = {
			add_building_construction = {
				type = nuclear_facility
				level = 1
				instant_build = yes
				province = 9292
			}
		}

		generate_scientist_character = {
			skills = {
				specialization_nuclear = 2
			}
			traits = { scientist_trait_nuclear_specialist }
		}
	}

	# --------------------------------------------------------------------- air
	option = {
		name = wif_sme.6.b

		572 = {
			add_building_construction = {
				type = air_facility
				level = 1
				instant_build = yes
				province = 9292
			}
		}

		generate_scientist_character = {
			skills = {
				specialization_air = 2
			}
			traits = { scientist_trait_aircraft_specialist }
		}
	}

	# -------------------------------------------------------------------- land
	option = {
		name = wif_sme.6.c

		572 = {
			add_building_construction = {
				type = land_facility
				level = 1
				instant_build = yes
				province = 9292
			}
		}

		generate_scientist_character = {
			skills = {
				specialization_land = 2
			}
			traits = { scientist_trait_armored_specialist }
		}
	}

	# ------------------------------------------------------------ all research
	#
	# The only option that builds nothing and grants no scientist. What it
	# leaves behind is a STATE flag and the dynamic modifier keyed to it - see
	# common/dynamic_modifiers/wif_chelyabinsk_institute.txt for why the bonus
	# cannot simply be a state modifier, and why the flag lives on the state
	# while the modifier lives on the country.
	#
	# The option name is .d and the description key is .desc rather than .d,
	# which is not an accident: a four-option event whose desc key were .d would
	# have the fourth button and the event text fighting over the same string.
	option = {
		name = wif_sme.6.d

		572 = {
			set_state_flag = wif_chelyabinsk_institute
		}

		# Granted here so it lands the same day rather than waiting for the
		# weekly pass. That pass is what hands it on afterwards.
		if = {
			limit = {
				NOT = { has_dynamic_modifier = { modifier = wif_chelyabinsk_institute } }
			}
			add_dynamic_modifier = { modifier = wif_chelyabinsk_institute }
		}
	}
}
