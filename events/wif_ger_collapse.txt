# The collapse of the Reich. Grand Funeral, Schemes of Power
# (common/national_focus/wif_ger_victory.txt) fires .1 the day after it
# completes. Each declaration marks and demilitarises a faction's regions;
# two days after the last, .6 splits the Reich, and ten days later .7 starts
# the war. Effects: common/scripted_effects/zz_wif_ger_collapse.txt.

# SS Gather in the Rhine.
country_event = {
	id = wif_ger_collapse.1
	title = wif_ger_collapse.1.t
	desc = wif_ger_collapse.1.d
	picture = GFX_report_event_german_troops

	is_triggered_only = yes

	immediate = {
		hidden_effect = { wif_cw_declare_ss = yes }
	}

	option = {
		name = wif_ger_collapse.1.a
		custom_effect_tooltip = wif_ger_collapse.1.a_tt
		hidden_effect = { country_event = { id = wif_ger_collapse.2 days = 3 } }
	}
}

	}

# Collapse of the Reich.
country_event = {
	id = wif_ger_collapse.6
	title = wif_ger_collapse.6.t
	desc = wif_ger_collapse.6.d
	picture = GFX_report_event_ger_civil_war

	is_triggered_only = yes

	# One option per side: the player goes to the side they pick
	# (wif_cw_switch_player). An option only shows if that side will have land.
	# The AI takes the first; for the AI the choice changes nothing.
	option = {
		name = wif_ger_collapse.6.bor
		ai_chance = { base = 1 }
		custom_effect_tooltip = wif_ger_collapse.6.a_tt
		custom_effect_tooltip = wif_ger_collapse.6.bor_tt
		hidden_effect = { wif_cw_collapse_and_war = yes }
	}
	option = {
		name = wif_ger_collapse.6.ssr
		trigger = { any_owned_state = { has_state_flag = wif_cw_ss } }
		ai_chance = { base = 0 }
		custom_effect_tooltip = wif_ger_collapse.6.a_tt
		custom_effect_tooltip = wif_ger_collapse.6.ssr_tt
		hidden_effect = {
			set_country_flag = wif_cw_play_ssr
			wif_cw_collapse_and_war = yes
		}
	}
	option = {
		name = wif_ger_collapse.6.spg
		trigger = { any_owned_state = { has_state_flag = wif_cw_speer } }
		ai_chance = { base = 0 }
		custom_effect_tooltip = wif_ger_collapse.6.a_tt
		custom_effect_tooltip = wif_ger_collapse.6.spg_tt
		hidden_effect = {
			set_country_flag = wif_cw_play_spg
			wif_cw_collapse_and_war = yes
		}
	}
	option = {
		name = wif_ger_collapse.6.gor
		trigger = { any_owned_state = { has_state_flag = wif_cw_goring } }
		ai_chance = { base = 0 }
		custom_effect_tooltip = wif_ger_collapse.6.a_tt
		custom_effect_tooltip = wif_ger_collapse.6.gor_tt
		hidden_effect = {
			set_country_flag = wif_cw_play_gor
			wif_cw_collapse_and_war = yes
		}
	}
	option = {
		name = wif_ger_collapse.6.dwm
		trigger = { any_owned_state = { has_state_flag = wif_cw_wehrmacht } }
		ai_chance = { base = 0 }
		custom_effect_tooltip = wif_ger_collapse.6.a_tt
		custom_effect_tooltip = wif_ger_collapse.6.dwm_tt
		hidden_effect = {
			set_country_flag = wif_cw_play_dwm
			wif_cw_collapse_and_war = yes
		}
	}
}
