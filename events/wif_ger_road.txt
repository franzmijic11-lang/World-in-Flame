# Gestapo Files on the General Staff.
country_event = {
	id = wif_ger_road.4
	title = wif_ger_road.4.t
	desc = wif_ger_road.4.d
	picture = GFX_report_event_bunker_01

	is_triggered_only = yes

	# Arrest three of the named generals.
	option = {
		name = wif_ger_road.4.a
		ai_chance = { factor = 50 }
		add_political_power = 50
		custom_effect_tooltip = wif_ger_road.4.a_tt
		hidden_effect = {
			random_unit_leader = {
				limit = {
					is_army_leader = yes
					NOT = { wif_is_ss_general = yes }
				}
				retire = yes
			}
			random_unit_leader = {
				limit = {
					is_army_leader = yes
					NOT = { wif_is_ss_general = yes }
				}
				retire = yes
			}
			random_unit_leader = {
				limit = {
					is_army_leader = yes
					NOT = { wif_is_ss_general = yes }
				}
				retire = yes
			}
		}
		custom_effect_tooltip = wif_ger_distrust_plus_15_tt
		hidden_effect = {
			set_temp_variable = { wif_dist = 15 }
			wif_ger_add_distrust = yes
		}
	}

	# Keep the files closed - for now.
	option = {
		name = wif_ger_road.4.b
		ai_chance = { factor = 50 }
		custom_effect_tooltip = wif_ger_distrust_plus_5_tt
		hidden_effect = {
			set_temp_variable = { wif_dist = 5 }
			wif_ger_add_distrust = yes
		}
	}
}
