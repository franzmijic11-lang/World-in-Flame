# Console shortcuts. Type in the console, as any country.
# (wif_debug.1-4 belong to the Soviet surrender tests in wif_debug_events.txt.)
#   event wif_debug.11   the German civil war: Fuhrer Found Dead, then the
#                       collapse chain (SS Gather in the Rhine onward)
#   event wif_debug.12   Contact Lost with Germania: the Ostschild forms and
#                       the SME's war begins
#   event wif_debug.13   the SME collapses (loses its war, or collapses outright
#                       if there is no war)
#   event wif_debug.14   the SME wins its war
#   event wif_debug.15   Yagoda rises: the West Siberian Marxists break away
#                       (fires the real wif_sme.4 on the SME, so an SME player
#                       still chooses between Moscow and Yagoda)
#   event wif_debug.16   repairs Zhukov's portraits in a running game (he showed
#                       the unknown portrait as ZWF's leader)
#   event wif_debug.17   Japan's Pacific settlement ("America out of Asia")
#                       right now, without the focus or its conditions
# ---------------------------------------------------------------------------

country_event = {
	id = wif_debug.11
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { country_exists = GER }
			GER = {
				if = {
					limit = { NOT = { has_country_flag = wif_ger_fuhrer_dead } }
					wif_ger_fuhrer_found_dead = yes
				}
				country_event = { id = wif_ger_collapse.1 days = 1 }
			}
		}
	}
}
