add_namespace = wif_usa_peace

# Germany offers the United States a separate peace after Britain and France
# have fallen. Fired from GER_crossing_the_atlantic, renamed "Force the
# Americans to see Reason".
country_event = {
	id = wif_usa_peace.1
	title = wif_usa_peace.1.t
	desc = wif_usa_peace.1.desc
	picture = GFX_report_event_german_politician_speech

	is_triggered_only = yes

	# Accept - leave the war
	option = {
		name = wif_usa_peace.1.a

		ai_chance = {
			base = 90

			# The worse the position in Europe looks, the more attractive a
			# settlement is.
			modifier = {
				factor = 2
				ENG = { has_capitulated = yes }
				FRA = { has_capitulated = yes }
			}
			modifier = {
				factor = 2
				NOT = { has_war_support > 0.5 }
			}
			modifier = { # A country already bleeding badly wants out
				factor = 2
				surrender_progress > 0.1
			}
		}

		# Other American majors settle on the same terms. Done BEFORE this
		# country leaves, while faction membership is still intact to test
		# against.
		every_other_country = {
			limit = {
				is_major = yes
				has_war_with = GER
				is_in_faction_with = ROOT
				OR = {
					capital_scope = { is_on_continent = north_america }
					capital_scope = { is_on_continent = south_america }
				}
			}
			wif_settle_with_axis = yes
		}

		wif_settle_with_axis = yes

		GER = {
			country_event = { id = wif_usa_peace.2 }
		}
	}

	# Decline - fight on
	option = {
		name = wif_usa_peace.1.b

		ai_chance = {
			base = 10

			modifier = { # Still confident, keep fighting
				factor = 3
				has_war_support > 0.7
			}
		}

		add_war_support = 0.05

		GER = {
			country_event = { id = wif_usa_peace.3 }
		}
	}
}

# Germany is told the Americans accepted
country_event = {
	id = wif_usa_peace.2
	title = wif_usa_peace.2.t
	desc = wif_usa_peace.2.desc
	picture = GFX_report_event_generic_signing

	is_triggered_only = yes

	option = {
		name = wif_usa_peace.2.a
		add_stability = 0.05
		add_war_support = 0.05
	}
}

# Germany is told the Americans refused.
#
# The refusal is what opens the American war. It delivers what the three
# deleted focuses under Force the Americans to see Reason used to give:
#   GER_shatter_usas_hegemony              - the timed spirit
#   GER_challenge_the_monroe_doctrine      - the Monroe destabilisation
#   GER_establish_protectorates_in_america - the South American claims and
#                                            the American Reichskommissariats
# The flag GER_americans_refused_peace is what common/decisions/GER.txt now
# checks in place of the deleted focuses.
country_event = {
	id = wif_usa_peace.3
	title = wif_usa_peace.3.t
	desc = wif_usa_peace.3.desc
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes

	option = {
		name = wif_usa_peace.3.a

		set_country_flag = GER_americans_refused_peace

		# Shatter USA's Hegemony
		add_timed_idea = {
			idea = GER_shatter_USA_hegemony_ns
			days = 365
		}

		# Challenge the Monroe Doctrine. Skipped if GER_disrupt_monroe already
		# gave the same package in peacetime, or Germany would get a second
		# Johannes Becker. GER_disrupt_monroe in turn is closed off once this
		# flag is set.
		if = {
			limit = {
				NOT = { has_completed_focus = GER_disrupt_monroe }
			}
			GER_setup_monroe_doctrine_countries_to_destabilize_array = yes
			set_country_flag = GER_destabilize_monroe_flag
			custom_effect_tooltip = GER_challenge_the_monroe_doctrine_tt

			hidden_effect = {
				if = {
					limit = {
						USA = {
							OR = {
								has_government = democratic
								has_government = communism
							}
						}
					}
					USA = {
						country_event = {
							id = wuw_GER_monroe.1
							hours = 12
						}
					}
				}
			}

			if = {
				limit = {
					has_dlc = "La Resistance"
				}
				GER_add_or_modify_military_intelligence_service = yes
				add_to_variable = {
					GER_military_intelligence_service_operative_slot = 1
					tooltip = operative_slot_tt
				}
				GER_create_operative_johannes_becker = yes
			}
		}

		custom_effect_tooltip = generic_skip_one_line_tt

		# Establish Protectorates in America: claims on every South American
		# state held by a power from outside the continent.
		every_state = {
			limit = {
				is_on_continent = south_america
				OWNER = {
					capital_scope = {
						NOT = {
							is_on_continent = south_america
						}
					}
				}
				NOT = {
					OR = {
						is_claimed_by = ROOT
						is_core_of = ROOT
						is_owned_by = ROOT
					}
				}
			}
			add_claim_by = ROOT
		}

		unlock_decision_tooltip = GER_reichskommissariat_klein_venedig
		unlock_decision_tooltip = GER_reichskommissariat_anden
		unlock_decision_tooltip = GER_reichskommissariat_kolumbus
		unlock_decision_tooltip = GER_reichskommissariat_nordamerika
	}
}
