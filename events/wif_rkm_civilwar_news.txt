add_namespace = wif_rkm_civilwar

# The rising breaks out. Fired from wif_nsr_revolt the moment the breakaways are
# created, and broadcast to the world. Purely an announcement - every mechanical
# effect of the rising happens in the scripted effect that fires this.
news_event = {
	id = wif_rkm_civilwar.2
	title = wif_rkm_civilwar.2.t
	desc = wif_rkm_civilwar.2.desc
	picture = GFX_wif_moskowien_civil_war

	major = yes

	is_triggered_only = yes

	# Moskowien
	option = {
		name = wif_rkm_civilwar.2.a
		trigger = {
			tag = RKM
		}
	}

	# Berlin
	option = {
		name = wif_rkm_civilwar.2.b
		trigger = {
			tag = GER
		}
	}

	# Everyone else. NOT wrapping an OR, not two tags in one NOT - the latter
	# reads as NOT(RKM and GER), which is true for everybody including Germany.
	option = {
		name = wif_rkm_civilwar.2.c
		trigger = {
			NOT = {
				OR = {
					tag = RKM
					tag = GER
				}
			}
		}
	}
}

# The Moskowien civil war is over. Fired from on_weekly once every breakaway is
# gone, and it is this event that carries out the reunification.
#
# The work sits in immediate, wrapped in RKM = { } so it runs on Moskowien no
# matter who is reading the news, and guarded on wif_moskowien_civil_war so it
# is idempotent. major = yes broadcasts it, and without that guard the
# annexation would be attempted once per recipient.
news_event = {
	id = wif_rkm_civilwar.1
	title = wif_rkm_civilwar.1.t
	desc = wif_rkm_civilwar.1.desc
	picture = GFX_wif_moskowien_civil_war

	major = yes

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			RKM = {
				if = {
					limit = { has_country_flag = wif_moskowien_civil_war }

					# Each loyalist folds back in separately, so one of them
					# being overrun during the war does not block the other.
					if = {
						limit = { country_exists = NSP }
						annex_country = {
							target = NSP
							transfer_troops = no
						}
						# Their own core, passed to Moskowien with the city.
						add_state_core = 195
					}
					if = {
						limit = { country_exists = NAR }
						annex_country = {
							target = NAR
							transfer_troops = no
						}
						add_state_core = 214
					}
					if = {
						limit = { country_exists = NWG }
						annex_country = {
							target = NWG
							transfer_troops = no
						}
						add_state_core = 217
					}
					if = {
						limit = { country_exists = LOK }
						annex_country = {
							target = LOK
							transfer_troops = no
						}
						add_state_core = 224
					}

					# The northern war ends with the civil war that caused it.
					#
					# If Archangelsk is still standing when Moskowien wins, the
					# SME has run out of the moment it was exploiting: it takes
					# a white peace and hands back the three states it took as
					# the risings began. wif_sme_eastern_annexation grabbed
					# Gorky, Kargopol and Kotlas without a declaration, on the
					# argument that the administration had stopped being able to
					# hold them. That argument no longer holds.
					#
					# If Archangelsk has already fallen, none of this runs - the
					# SME keeps it, and the NAR fold-back above is skipped by
					# its own country_exists check.
					if = {
						limit = {
							country_exists = NAR
							country_exists = SME
							SME = { has_war_with = NAR }
						}
						SME = {
							white_peace = NAR

							252 = { transfer_state_to = RKM }
							879 = { transfer_state_to = RKM }
							880 = { transfer_state_to = RKM }
						}
						add_state_core = 252
						add_state_core = 879
						add_state_core = 880
					}

					# Moskau itself, if it is somehow still not core.
					#
					# The normal route is the stabilisation chain, which cores
					# 219 in wif_rkm.10 - but that only fires if Berlin agreed
					# to fund the operation, and Berlin can refuse. A
					# Reichskommissariat that has just held the city through a
					# civil war has settled the question by other means.
					if = {
						limit = {
							NOT = { 219 = { is_core_of = RKM } }
						}
						add_state_core = 219
					}

					clr_country_flag = wif_moskowien_civil_war

					# The emergency surrender limit only ever applied while the
					# risings held Moskowien's cities. Granted in wif_nsr_revolt.
					remove_ideas = wif_rkm_civil_war_emergency
				}
			}
		}
	}

	# Moskowien's own view of it
	option = {
		name = wif_rkm_civilwar.1.a
		trigger = {
			tag = RKM
		}
	}

	# Everyone else
	option = {
		name = wif_rkm_civilwar.1.b
		trigger = {
			NOT = { tag = RKM }
		}
	}
}
